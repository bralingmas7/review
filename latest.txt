   108	  Future<T> _withDriveApi<T>(
   109	    Future<T> Function(drive.DriveApi api) action,
   110	  ) async {
   111	    final currentAccount = await restoreAccount();
   112	
   113	    if (currentAccount == null) {
   114	      throw StateError('Hubungkan akun Google Drive terlebih dahulu.');
   115	    }
   116	
   117	    final authorizationClient = currentAccount.authorizationClient;
   118	    final authorization =
   119	        await authorizationClient.authorizationForScopes(_scopes) ??
   120	        await authorizationClient.authorizeScopes(_scopes);
   121	
   122	    final client = authorization.authClient(scopes: _scopes);
   123	
   124	    try {
   125	      final api = drive.DriveApi(client);
   126	      return await action(api);
   127	    } finally {
   128	      client.close();
   129	    }
   130	  }

   132	  void _validateEncryptedFile(List<int> bytes) {
   133	    if (bytes.isEmpty || bytes.length > _maximumBackupBytes) {
   134	      throw const FormatException('Ukuran backup tidak valid.');
   135	    }
   136	
   137	    try {
   138	      final decoded = jsonDecode(utf8.decode(bytes));
   139	
   140	      if (decoded is! Map<String, dynamic> ||
   141	          decoded['format'] != _backupFormat ||
   142	          decoded['version'] != _backupVersion ||
   143	          decoded['cipher'] != 'AES-256-GCM' ||
   144	          decoded['kdf'] != 'PBKDF2-HMAC-SHA256' ||
   145	          decoded['iterations'] != 310000 ||
   146	          decoded['salt'] is! String ||
   147	          decoded['nonce'] is! String ||
   148	          decoded['ciphertext'] is! String ||
   149	          decoded['mac'] is! String) {
   150	        throw const FormatException(
   151	          'File bukan backup terenkripsi Buku Utang Warung.',
   152	        );
   153	      }
   154	    } on FormatException {
   155	      rethrow;
   156	    } catch (_) {
   157	      throw const FormatException('Isi file backup tidak valid.');
   158	    }
   159	  }

   169	      final metadata = drive.File()

   172	        ..appProperties = {
   173	          'app': 'buku_utang_warung',
   174	          'format': 'encrypted_backup',
   175	          'version': '1',
   176	        };
   177	
   178	      final media = drive.Media(
   179	        Stream<List<int>>.value(encryptedBytes),
   180	        encryptedBytes.length,
   181	      );
   182	
   183	      final uploaded = await api.files.create(
   184	        metadata,
   185	        uploadMedia: media,
   186	        $fields: 'id,name,mimeType,size,createdTime',
   187	      );

   203	        final response = await api.files.list(
   204	          q: "trashed = false and "
   205	              "appProperties has { key='app' and "
   206	              "value='buku_utang_warung' }",
   207	          spaces: 'drive',
   208	          pageSize: 100,
   209	          pageToken: pageToken,
   210	          orderBy: 'createdTime desc',
   211	          $fields: 'nextPageToken,files(id,name,mimeType,size,createdTime)',
   212	        );
   213	
   214	        results.addAll(response.files ?? <drive.File>[]);
   215	        pageToken = response.nextPageToken;
   216	      } while (pageToken != null);

   222	  Future<Uint8List> downloadEncryptedBackup(String fileId) async {
   223	    if (fileId.trim().isEmpty) {
   224	      throw ArgumentError('ID file backup kosong.');
   225	    }
   226	
   227	    return _withDriveApi((api) async {
   228	      final response = await api.files.get(
   229	        fileId,
   230	        downloadOptions: drive.DownloadOptions.fullMedia,
   231	      );
   232	
   233	      if (response is! drive.Media) {
   234	        throw StateError('Google Drive tidak mengembalikan isi file.');
   235	      }
   236	
   237	      final builder = BytesBuilder(copy: false);
   238	
   239	      await for (final chunk in response.stream) {
   240	        builder.add(chunk);
   241	
   242	        if (builder.length > _maximumBackupBytes) {
   243	          throw const FormatException(
   244	            'File backup melebihi batas ukuran 100 MB.',
   245	          );
   246	        }
   247	      }
   248	
   249	      final bytes = builder.takeBytes();
   250	      _validateEncryptedFile(bytes);
   251	
   252	      return bytes;
   253	    });
   254	  }
