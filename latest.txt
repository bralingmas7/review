import 'package:flutter/material.dart';
import 'package:flutter/services.dart';

class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key});

  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> {
  List<CustomerSummary> _items = [];
  bool _loading = true;
  String _query = '';

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<void> _openSettings() async {
    await Navigator.push(
      context,
      MaterialPageRoute(builder: (_) => const SettingsScreen()),
    );
    await _load();
  }

  Future<void> _tagihCepat(CustomerSummary s) async {
    final c = s.customer;
    final txns = await AppDb.instance.txnsFor(c.id!);
    final cycle = activeDebtCycle(txns);
    final msg = buildReminder(
      name: c.name,
      balance: computeBalance(txns),
      due: nextDueDate(cycle),
      txns: cycle,
    );

    bool ok;
    try {
      ok = await launchUrl(
        waUri(c.phone, msg),
        mode: LaunchMode.externalApplication,
      );
    } catch (_) {
      ok = false;
    }

    if (!ok && mounted) {
      await Clipboard.setData(ClipboardData(text: msg));
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'WhatsApp tidak bisa dibuka. Pesan disalin, tempel manual ya.',
          ),
        ),
      );
    }
  }

  Future<void> _load() async {
    final db = AppDb.instance;
    final customers = await db.customers();
    await BackupStatus.instance.load(hasData: customers.isNotEmpty);
    final txns = await db.allTxns();
    final list = buildSummaries(customers, txns)
      ..sort((a, b) {
        final byBalance = b.balance.compareTo(a.balance);
        return byBalance != 0
            ? byBalance
            : a.customer.name
                .toLowerCase()
                .compareTo(b.customer.name.toLowerCase());
      });
    if (!mounted) return;
    setState(() {
      _items = list;
      _loading = false;
    });
  }

  Future<Customer?> _createCustomer() async {
    final c = await showCustomerForm(context);
    if (c == null) return null;
    final id = await AppDb.instance.insertCustomer(c);
    return c.copyWith(id: id);
  }

  Future<void> _addCustomerOnly() async {
    final c = await _createCustomer();
    if (c != null) await _load();
  }

  /// Alur cepat: pilih pelanggan -> isi nominal -> simpan.
  Future<void> _quickDebt() async {
    final picked = await showModalBottomSheet<Object>(
      context: context,
      isScrollControlled: true,
      builder: (_) => PickCustomerSheet(items: _items),
    );
    if (picked == null || !mounted) return;

    Customer? customer;
    var balance = 0;
    if (picked is CustomerSummary) {
      customer = picked.customer;
      balance = picked.balance;
    } else {
      customer = await _createCustomer();
    }
    if (customer == null || !mounted) return;

    final txn = await showTxnForm(
      context,
      customer: customer,
      type: TxnType.utang,
      balance: balance,
    );
    if (txn != null) await AppDb.instance.insertTxn(txn);
    await _load();
  }

  Future<void> _openCustomer(Customer c) async {
    await Navigator.push(
      context,
      MaterialPageRoute(builder: (_) => CustomerScreen(customerId: c.id!)),
    );
    await _load();
  }

  @override
  Widget build(BuildContext context) {
    final q = _query.trim().toLowerCase();
    final visible = _items
        .where((s) => q.isEmpty || s.customer.name.toLowerCase().contains(q))
        .toList();

    final totalPiutang =
        _items.fold<int>(0, (sum, s) => sum + (s.balance > 0 ? s.balance : 0));
    final telat =
        _items.where((s) => s.status() == DebtStatus.telat).length;
    final mendekati =
        _items.where((s) => s.status() == DebtStatus.mendekati).length;

    return Scaffold(
      appBar: AppBar(
        title: const Text('Buku Utang Warung'),
        actions: [
          BackupIndicator(onTap: _openSettings),
          IconButton(
            tooltip: 'Pengaturan',
            icon: const Icon(Icons.settings_outlined),
            onPressed: () async {
              await Navigator.push(
                context,
                MaterialPageRoute(
                  builder: (_) => const SettingsScreen(),
                ),
              );
              await _load();
            },
          ),
          IconButton(
            tooltip: 'Pelanggan baru',
            icon: const Icon(Icons.person_add_alt_1),
            onPressed: _addCustomerOnly,
          ),
        ],
      ),
      floatingActionButton: FloatingActionButton.extended(
        onPressed: _quickDebt,
        icon: const Icon(Icons.add),
        label: const Text('Utang baru'),
      ),
      body: _loading
          ? const Center(child: CircularProgressIndicator())
          : Column(
              children: [
                _SummaryCard(
                  total: totalPiutang,
                  telat: telat,
                  mendekati: mendekati,
                ),
                Padding(
                  padding: const EdgeInsets.fromLTRB(16, 0, 16, 8),
                  child: TextField(
                    decoration: const InputDecoration(
                      prefixIcon: Icon(Icons.search),
                      hintText: 'Cari pelanggan',
                      border: OutlineInputBorder(),
                      isDense: true,
                    ),
                    onChanged: (v) => setState(() => _query = v),
                  ),
                ),
                Expanded(
                  child: visible.isEmpty
                      ? Center(
                          child: Padding(
                            padding: const EdgeInsets.all(32),
                            child: Text(
                              _items.isEmpty
                                  ? 'Belum ada pelanggan.\nKetuk "Utang baru" untuk mulai mencatat.'
                                  : 'Tidak ada pelanggan dengan nama itu.',
                              textAlign: TextAlign.center,
                            ),
                          ),
                        )
                      : ListView.builder(
                          padding: const EdgeInsets.only(bottom: 88),
                          itemCount: visible.length,
                          itemBuilder: (_, i) => _CustomerTile(
                            summary: visible[i],
                            onTap: () => _openCustomer(visible[i].customer),
                            onTagih: () => _tagihCepat(visible[i]),
                          ),
                        ),
                ),
              ],
            ),
    );
  }
}

          if (status == DebtStatus.telat ||
              status == DebtStatus.mendekati)

              tooltip: 'Tagih lewat WhatsApp',

import 'dart:async';
import 'dart:convert';
import 'dart:typed_data';

import 'package:file_picker/file_picker.dart';
import 'package:flutter/material.dart';
import 'package:google_sign_in/google_sign_in.dart';

import '../db.dart';
import '../services/backup_crypto_service.dart';

import '../services/google_drive_service.dart';

class SettingsScreen extends StatefulWidget {
  const SettingsScreen({super.key});

  @override
  State<SettingsScreen> createState() => _SettingsScreenState();
}

class _SettingsScreenState extends State<SettingsScreen> {
  bool _busy = false;
  String? _googleEmail;
  String _googleRestoreStatus = 'Belum diperiksa';
  StreamSubscription<GoogleSignInAccount?>? _googleAccountSubscription;

  @override
  void initState() {
    super.initState();

    final google = GoogleDriveService.instance;
    _googleEmail = google.account?.email;

    _googleAccountSubscription = google.accountChanges.listen(
      (account) {
        // Event tanpa akun tidak menghapus identitas yang tersimpan.
        // Logout eksplisit ditangani oleh _logoutGoogle().
        if (!mounted || account == null) return;

        setState(() {
          _googleEmail = account.email;
          _googleRestoreStatus =
              GoogleDriveService.instance.restoreDiagnostic;
        });
      },
      onError: (Object error, StackTrace stack) {
        debugPrint('Perubahan sesi Google gagal: $error');
      },
    );

    _restoreGoogleStatus();
  }

  @override
  void dispose() {
    _googleAccountSubscription?.cancel();
    super.dispose();
  }

  Future<void> _restoreGoogleStatus() async {
    try {
      final google = GoogleDriveService.instance;
      final account = await google.restoreAccount();
      final savedEmail = await google.loadSavedAccountEmail();

      if (!mounted) return;

      setState(() {
        _googleEmail = account?.email ?? savedEmail;
        final base = account != null
            ? 'SESI_AKTIF'
            : savedEmail != null
                ? 'IDENTITAS_TERSIMPAN_SESI_BELUM_AKTIF'
                : 'BELUM_TERHUBUNG';
        _googleRestoreStatus = '$base | ${google.restoreDiagnostic}';
      });
    } catch (e) {
      if (!mounted) return;
      setState(() {
        _googleRestoreStatus =
            GoogleDriveService.instance.restoreDiagnostic;
      });
      debugPrint('Pemulihan status Google gagal: $e');
    }
  }

  Future<void> _connectGoogle() async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      final email =
          await GoogleDriveService.instance.connectAndTest();

      if (!mounted) return;
      setState(() => _googleEmail = email);
      _message('Google Drive berhasil terhubung: $email');
    } catch (e) {
      _message('Koneksi Google gagal: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<void> _switchGoogle() async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      final email =
          await GoogleDriveService.instance.switchAccount();

      if (!mounted) return;
      setState(() => _googleEmail = email);
      _message('Akun Google berhasil diganti: $email');
    } catch (e) {
      _message('Ganti akun gagal: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<void> _logoutGoogle() async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      await GoogleDriveService.instance.signOut();

      if (!mounted) return;
      setState(() => _googleEmail = null);
      _message('Logout Google berhasil. Data lokal tetap aman.');
    } catch (e) {
      _message('Logout Google gagal: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  void _message(String message) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(message)),
    );
  }

  Future<void> _backup() async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      final revision = BackupStatus.instance.revision;
      final data = await AppDb.instance.exportBackup();
      final json = const JsonEncoder.withIndent('  ').convert(data);
      final now = DateTime.now();
      final stamp = '${now.year}'
          '${now.month.toString().padLeft(2, '0')}'
          '${now.day.toString().padLeft(2, '0')}_'
          '${now.hour.toString().padLeft(2, '0')}'
          '${now.minute.toString().padLeft(2, '0')}';

      final result = await FilePicker.saveFile(
        dialogTitle: 'Simpan backup Buku Utang Warung',
        fileName: 'backup_buku_utang_$stamp.json',
        bytes: Uint8List.fromList(utf8.encode(json)),
        mimeType: 'application/json',
      );

      if (result != null) {
        await BackupStatus.instance.markLocalBackedUp(revision);
        _message('File backup berhasil disimpan.');
      }
    } catch (e) {
      _message('Backup gagal: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<void> _restore() async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      final file = await FilePicker.pickFile(
        type: FileType.custom,
        allowedExtensions: ['json'],
      );

      if (file == null) return;

      final bytes = await file.readAsBytes();
      final decoded = jsonDecode(utf8.decode(bytes));

      if (decoded is! Map<String, dynamic>) {
        throw const FormatException('Isi file JSON tidak valid');
      }

      // Periksa semua isi file sebelum meminta persetujuan penggantian data.
      AppDb.validateBackupData(decoded);

      final customerCount = (decoded['customers'] as List).length;
      final txnCount = (decoded['transactions'] as List).length;

      if (!mounted) return;

      final confirmed = await showDialog<bool>(
        context: context,
        builder: (context) => AlertDialog(
          title: const Text('Pulihkan data?'),
          content: Text(
            'Data saat ini akan DIGANTI dengan isi backup.\n\n'
            'Pelanggan: $customerCount\n'
            'Transaksi: $txnCount\n\n'
            'Pastikan Anda sudah membuat backup data saat ini.',
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(context, false),
              child: const Text('Batal'),
            ),
            FilledButton(
              onPressed: () => Navigator.pop(context, true),
              child: const Text('Ganti dan pulihkan'),
            ),
          ],
        ),
      );

      if (confirmed != true || !mounted) return;

      await AppDb.instance.restoreBackup(decoded);
      _message('Restore berhasil. Data telah dipulihkan.');
    } catch (e) {
      _message('Restore gagal. Data lama tetap aman: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }


  Future<String?> _askCloudPassword({required bool confirm}) async {
    final passwordController = TextEditingController();
    final confirmController = TextEditingController();
    final formKey = GlobalKey<FormState>();
    String? mismatch;

    try {
      return await showDialog<String>(
        context: context,
        barrierDismissible: false,
        builder: (dialogContext) => StatefulBuilder(
          builder: (dialogContext, setDialogState) => AlertDialog(
            title: Text(
              confirm ? 'Buat password backup' : 'Password backup',
            ),
            content: Form(
              key: formKey,
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  TextFormField(
                    controller: passwordController,
                    obscureText: true,
                    autofocus: true,
                    decoration: const InputDecoration(
                      labelText: 'Password backup',
                      helperText: 'Minimal 8 karakter',
                    ),
                    validator: (value) {
                      if ((value ?? '').runes.length < 8) {
                        return 'Password minimal 8 karakter.';
                      }
                      return null;
                    },
                  ),
                  if (confirm) ...[
                    const SizedBox(height: 12),
                    TextField(
                      controller: confirmController,
                      obscureText: true,
                      decoration: const InputDecoration(
                        labelText: 'Ulangi password backup',
                      ),
                      onChanged: (_) {
                        if (mismatch != null) {
                          setDialogState(() => mismatch = null);
                        }
                      },
                    ),
                  ],
                  if (mismatch != null) ...[
                    const SizedBox(height: 8),
                    Text(
                      mismatch!,
                      style: TextStyle(
                        color: Theme.of(dialogContext).colorScheme.error,
                      ),
                    ),
                  ],
                  const SizedBox(height: 8),
                  const Text(
                    'Password ini berbeda dari password Google. '
                    'Aplikasi tidak menyimpan password ini. '
                    'Jika lupa, backup tidak dapat dipulihkan.',
                    style: TextStyle(fontSize: 12),
                  ),
                ],
              ),
            ),
            actions: [
              TextButton(
                onPressed: () => Navigator.pop(dialogContext),
                child: const Text('Batal'),
              ),
              FilledButton(
                onPressed: () {
                  if (formKey.currentState?.validate() != true) return;

                  if (confirm &&
                      passwordController.text != confirmController.text) {
                    setDialogState(
                      () => mismatch = 'Password yang dimasukkan tidak sama.',
                    );
                    return;
                  }

                  Navigator.pop(dialogContext, passwordController.text);
                },
                child: const Text('Lanjutkan'),
              ),
            ],
          ),
        ),
      );
    } finally {
      passwordController.dispose();
      confirmController.dispose();
    }
  }

  Future<void> _cloudBackup() async {
    if (_busy || _googleEmail == null) return;

    setState(() => _busy = true);

    try {
      if (GoogleDriveService.instance.account == null) {
        final email = await GoogleDriveService.instance.connectAndTest();
        if (!mounted) return;
        setState(() => _googleEmail = email);
      }

      final password = await _askCloudPassword(confirm: true);
      if (password == null || !mounted) return;

      final revision = BackupStatus.instance.revision;
      final data = await AppDb.instance.exportBackup();
      final encrypted =
          await BackupCryptoService.instance.encryptBackup(data, password);

      final uploaded =
          await GoogleDriveService.instance.uploadEncryptedBackup(encrypted);

      await BackupStatus.instance.markBackedUp(revision);
      _message('Backup terenkripsi berhasil diunggah: ${uploaded.name}');
    } catch (e) {
      _message('Backup Google Drive gagal: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<void> _cloudRestore() async {
    if (_busy || _googleEmail == null) return;

    setState(() => _busy = true);

    try {
      if (GoogleDriveService.instance.account == null) {
        final email = await GoogleDriveService.instance.connectAndTest();
        if (!mounted) return;
        setState(() => _googleEmail = email);
      }

      final files =
          await GoogleDriveService.instance.listEncryptedBackups();

      if (!mounted) return;

      if (files.isEmpty) {
        _message('Belum ada backup Buku Utang Warung di Google Drive.');
        return;
      }

      final selectedIndex = await showDialog<int>(
        context: context,
        builder: (dialogContext) => AlertDialog(
          title: const Text('Pilih backup cloud'),
          content: SizedBox(
            width: double.maxFinite,
            child: ListView(
              shrinkWrap: true,
              children: [
                for (var i = 0; i < files.length; i++)
                  ListTile(
                    leading: const Icon(Icons.cloud_download_outlined),
                    title: Text(files[i].name ?? 'Backup tanpa nama'),
                    subtitle: Text(
                      files[i].createdTime?.toLocal().toString() ??
                          'Waktu tidak tersedia',
                    ),
                    onTap: () => Navigator.pop(dialogContext, i),
                  ),
              ],
            ),
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text('Batal'),
            ),
          ],
        ),
      );

      if (selectedIndex == null || !mounted) return;

      final selectedFile = files[selectedIndex];
      final fileId = selectedFile.id;
      if (fileId == null || fileId.isEmpty) {
        throw StateError('ID backup Google Drive tidak tersedia.');
      }

      final password = await _askCloudPassword(confirm: false);
      if (password == null || !mounted) return;

      final encrypted =
          await GoogleDriveService.instance.downloadEncryptedBackup(fileId);

      final decoded =
          await BackupCryptoService.instance.decryptBackup(
        encrypted,
        password,
      );

      AppDb.validateBackupData(decoded);
