import 'dart:async';
import 'dart:convert';
import 'dart:typed_data';

import 'package:extension_google_sign_in_as_googleapis_auth/extension_google_sign_in_as_googleapis_auth.dart';
import 'package:google_sign_in/google_sign_in.dart';
import 'package:googleapis/drive/v3.dart' as drive;

class GoogleDriveService {
  GoogleDriveService._();

  static final GoogleDriveService instance = GoogleDriveService._();

  static const String _serverClientId =
      '83176740993-ughbfdiak2v3mq31prsfpae7pffkaku7.apps.googleusercontent.com';

  static const List<String> _scopes = [
    'https://www.googleapis.com/auth/drive.file',
  ];

  static const String _backupFormat =
      'buku_utang_warung_encrypted_backup';

  static const int _backupVersion = 1;
  static const int _maximumBackupBytes = 100 * 1024 * 1024;

  final GoogleSignIn _google = GoogleSignIn.instance;
  Future<void>? _initialization;
  GoogleSignInAccount? _account;

  GoogleSignInAccount? get account => _account;

  Future<void> initialize() {
    return _initialization ??= _google.initialize(
      serverClientId: _serverClientId,
    );
  }

  Future<GoogleSignInAccount?> restoreAccount() async {
    await initialize();

    if (_account != null) return _account;

    final attempt = _google.attemptLightweightAuthentication();

    if (attempt == null) return null;

    _account = await attempt;
    return _account;
  }

  Future<T> _withDriveApi<T>(
    Future<T> Function(drive.DriveApi api) action,
  ) async {
    final currentAccount = await restoreAccount();

    if (currentAccount == null) {
      throw StateError('Hubungkan akun Google Drive terlebih dahulu.');
    }

    final authorization = await currentAccount.authorizationClient
        .authorizeScopes(_scopes);

    final client = authorization.authClient(scopes: _scopes);

    try {
      final api = drive.DriveApi(client);
      return await action(api);
    } finally {
      client.close();
    }
  }

  void _validateEncryptedFile(List<int> bytes) {
    if (bytes.isEmpty || bytes.length > _maximumBackupBytes) {
      throw const FormatException('Ukuran backup tidak valid.');
    }

    try {
      final decoded = jsonDecode(utf8.decode(bytes));

      if (decoded is! Map<String, dynamic> ||
          decoded['format'] != _backupFormat ||
          decoded['version'] != _backupVersion ||
          decoded['cipher'] != 'AES-256-GCM' ||
          decoded['kdf'] != 'PBKDF2-HMAC-SHA256' ||
          decoded['salt'] is! String ||
          decoded['nonce'] is! String ||
          decoded['ciphertext'] is! String ||
          decoded['mac'] is! String) {
        throw const FormatException(
          'File bukan backup terenkripsi Buku Utang Warung.',
        );
      }
    } on FormatException {
      rethrow;
    } catch (_) {
      throw const FormatException('Isi file backup tidak valid.');
    }
  }

  Future<drive.File> uploadEncryptedBackup(
    List<int> encryptedBytes,
  ) async {
    _validateEncryptedFile(encryptedBytes);

    final timestamp = DateTime.now().toUtc().millisecondsSinceEpoch;

    return _withDriveApi((api) async {
      final metadata = drive.File()
        ..name = 'backup_buku_utang_$timestamp.buwbackup'
        ..mimeType = 'application/octet-stream'
        ..appProperties = {
          'app': 'buku_utang_warung',
          'format': 'encrypted_backup',
          'version': '1',
        };

      final media = drive.Media(
        Stream<List<int>>.value(encryptedBytes),
        encryptedBytes.length,
      );

      final uploaded = await api.files.create(
        metadata,
        uploadMedia: media,
        $fields: 'id,name,mimeType,size,createdTime',
      );

      if (uploaded.id == null) {
        throw StateError('Upload selesai tetapi ID file tidak ditemukan.');
      }

      return uploaded;
    });
  }

  Future<List<drive.File>> listEncryptedBackups() async {
    return _withDriveApi((api) async {
      final results = <drive.File>[];
      String? pageToken;

      do {
        final response = await api.files.list(
          q: "trashed = false and "
              "appProperties has { key='app' and "
              "value='buku_utang_warung' }",
          spaces: 'drive',
          pageSize: 100,
          pageToken: pageToken,
          orderBy: 'createdTime desc',
          $fields: 'nextPageToken,files(id,name,mimeType,size,createdTime)',
        );

        results.addAll(response.files ?? <drive.File>[]);
        pageToken = response.nextPageToken;
      } while (pageToken != null);

      return results;
    });
  }

  Future<Uint8List> downloadEncryptedBackup(String fileId) async {
    if (fileId.trim().isEmpty) {
      throw ArgumentError('ID file backup kosong.');
    }

    return _withDriveApi((api) async {
      final response = await api.files.get(
        fileId,
        downloadOptions: drive.DownloadOptions.fullMedia,
      );

      if (response is! drive.Media) {
        throw StateError('Google Drive tidak mengembalikan isi file.');
      }

      final builder = BytesBuilder(copy: false);

      await for (final chunk in response.stream) {
        builder.add(chunk);

        if (builder.length > _maximumBackupBytes) {
          throw const FormatException(
            'File backup melebihi batas ukuran 100 MB.',
          );
        }
      }

      final bytes = builder.takeBytes();
      _validateEncryptedFile(bytes);

      return bytes;
    });
  }

  Future<String> connectAndTest() async {
    await initialize();

    if (!_google.supportsAuthenticate()) {
      throw Exception('Metode login tidak didukung pada perangkat ini.');
    }

    final account = await _google.authenticate();
    _account = account;

    final authorization =
        await account.authorizationClient.authorizeScopes(_scopes);

    final client = authorization.authClient(scopes: _scopes);

    try {
      final api = drive.DriveApi(client);

      await api.files.list(
        pageSize: 1,
        $fields: 'files(id,name)',
      );
    } finally {
      client.close();
    }

    return account.email;
  }

  Future<void> signOut() async {
    await initialize();
    await _google.signOut();
    _account = null;
  }

  Future<String> switchAccount() async {
    await signOut();
    return connectAndTest();
  }
}

import 'dart:convert';
import 'dart:math';
import 'dart:typed_data';

import 'package:cryptography/cryptography.dart';

class BackupCryptoService {
  BackupCryptoService._();

  static final BackupCryptoService instance = BackupCryptoService._();

  static const int minimumPasswordLength = 8;
  static const int _version = 1;
  static const int _iterations = 310000;
  static const int _maximumEncryptedBytes = 100 * 1024 * 1024;

  static const String _format = 'buku_utang_warung_encrypted_backup';
  static const String _cipherName = 'AES-256-GCM';
  static const String _kdfName = 'PBKDF2-HMAC-SHA256';

  final Pbkdf2 _kdf = Pbkdf2(
    macAlgorithm: Hmac.sha256(),
    iterations: _iterations,
    bits: 256,
  );

  final AesGcm _cipher = AesGcm.with256bits();

  void _validatePassword(String password) {
    if (password.runes.length < minimumPasswordLength) {
      throw ArgumentError(
        'Password backup minimal $minimumPasswordLength karakter.',
      );
    }
  }

  Future<Uint8List> encryptBackup(
    Map<String, Object?> backup,
    String password,
  ) async {
    _validatePassword(password);

    final salt = Uint8List(16);
    final random = Random.secure();

    for (var i = 0; i < salt.length; i++) {
      salt[i] = random.nextInt(256);
    }

    final key = await _kdf.deriveKeyFromPassword(
      password: password,
      nonce: salt,
    );

    final clearBytes = utf8.encode(jsonEncode(backup));
    final encrypted = await _cipher.encrypt(
      clearBytes,
      secretKey: key,
    );

    final envelope = <String, Object?>{
      'format': _format,
      'version': _version,
      'cipher': _cipherName,
      'kdf': _kdfName,
      'iterations': _iterations,
      'salt': base64Encode(salt),
      'nonce': base64Encode(encrypted.nonce),
      'ciphertext': base64Encode(encrypted.cipherText),
      'mac': base64Encode(encrypted.mac.bytes),
    };

    return Uint8List.fromList(utf8.encode(jsonEncode(envelope)));
  }

  Future<Map<String, dynamic>> decryptBackup(
    List<int> encryptedBytes,
    String password,
  ) async {
    _validatePassword(password);

    if (encryptedBytes.isEmpty ||
        encryptedBytes.length > _maximumEncryptedBytes) {
      throw const FormatException('Ukuran file backup tidak valid.');
    }

    final decodedEnvelope = jsonDecode(utf8.decode(encryptedBytes));

    if (decodedEnvelope is! Map<String, dynamic>) {
      throw const FormatException('Format file backup tidak valid.');
    }

    final envelope = decodedEnvelope;

    if (envelope['format'] != _format ||
        envelope['version'] != _version ||
        envelope['cipher'] != _cipherName ||
        envelope['kdf'] != _kdfName ||
        envelope['iterations'] != _iterations) {
      throw const FormatException(
        'Format atau versi enkripsi backup tidak didukung.',
      );
    }

    try {
      final salt = base64Decode(envelope['salt'] as String);
      final nonce = base64Decode(envelope['nonce'] as String);
      final ciphertext = base64Decode(envelope['ciphertext'] as String);
      final macBytes = base64Decode(envelope['mac'] as String);

      if (salt.length != 16 ||
          nonce.length != 12 ||
          macBytes.length != 16 ||
          ciphertext.isEmpty) {
        throw const FormatException('Data enkripsi backup tidak lengkap.');
      }

      final key = await _kdf.deriveKeyFromPassword(
        password: password,
        nonce: salt,
      );

      final secretBox = SecretBox(
        ciphertext,
        nonce: nonce,
        mac: Mac(macBytes),
      );

      final clearBytes = await _cipher.decrypt(
        secretBox,
        secretKey: key,
      );

      final decodedBackup = jsonDecode(utf8.decode(clearBytes));

      if (decodedBackup is! Map<String, dynamic>) {
        throw const FormatException('Isi backup bukan objek JSON.');
      }

      return decodedBackup;
    } on FormatException {
      rethrow;
    } catch (_) {
      throw const FormatException(
        'Backup gagal dibuka. Password salah atau file telah berubah.',
      );
    }
  }
}
