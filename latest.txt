import 'models.dart';

const _bulan = [
  'Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun',
  'Jul', 'Agu', 'Sep', 'Okt', 'Nov', 'Des',
];

/// 15000 -> "Rp 15.000"
String rupiah(int value) {
  final digits = value.abs().toString();
  final buf = StringBuffer();
  for (var i = 0; i < digits.length; i++) {
    if (i > 0 && (digits.length - i) % 3 == 0) buf.write('.');
    buf.write(digits[i]);
  }
  return '${value < 0 ? '-' : ''}Rp $buf';
}

/// DateTime(2026, 11, 5) -> "5 Nov 2026"
String tglJam(DateTime d) {
  if (d.hour == 0 && d.minute == 0 && d.second == 0 && d.millisecond == 0) {
    return tgl(d);
  }
  String two(int n) => n.toString().padLeft(2, '0');
  return '${two(d.day)}-${two(d.month)}-${d.year} '
      '${two(d.hour)}:${two(d.minute)}:${two(d.second)}';
}

String tgl(DateTime d) => '${d.day} ${_bulan[d.month - 1]} ${d.year}';

String statusText(CustomerSummary s, [DateTime? now]) {
  final today = dateOnly(now ?? DateTime.now());
  switch (s.status(now)) {
    case DebtStatus.lunas:
      return 'Lunas';
    case DebtStatus.telat:
      final days = today.difference(dateOnly(s.nextDue!)).inDays;
      return 'Telat $days hari';
    case DebtStatus.mendekati:
      final days = dateOnly(s.nextDue!).difference(today).inDays;
      return days == 0 ? 'Jatuh tempo hari ini' : 'Jatuh tempo $days hari lagi';
    case DebtStatus.lancar:
      return s.nextDue == null ? 'Utang berjalan' : 'Tempo ${tgl(s.nextDue!)}';
  }
}

/// Ubah nomor lokal ke format internasional tanpa plus untuk wa.me.
/// "0812-3456-789" -> "628123456789". Mengembalikan null jika kosong.
String? normalizePhone(String raw) {
  final d = raw.replaceAll(RegExp(r'[^0-9]'), '');
  if (d.isEmpty) return null;
  if (d.startsWith('62')) return d;
  if (d.startsWith('0')) return '62${d.substring(1)}';
  if (d.startsWith('8')) return '62$d';
  return d;
}

/// Tautan WhatsApp. Tanpa nomor, WhatsApp membiarkan pengguna memilih kontak.
Uri waUri(String phone, String text) {
  final n = normalizePhone(phone);
  final encoded = Uri.encodeComponent(text);
  return Uri.parse('https://wa.me/${n ?? ''}?text=$encoded');
}

String _line(Txn t) {
  final sign = t.type == TxnType.utang ? '+' : '-';
  final label = t.type == TxnType.utang ? 'utang' : 'bayar';
  final note = t.note.trim().isEmpty ? '' : ' (${t.note.trim()})';
  return '${tglJam(t.date)}: $label $sign${rupiah(t.amount)}$note';
}

List<Txn> _newestFirst(List<Txn> txns) =>
    [...txns]..sort((a, b) => b.date.compareTo(a.date));

/// Pesan tagih yang sopan, siap kirim lewat WhatsApp.
String buildReminder({
  required String name,
  required int balance,
  DateTime? due,
  required List<Txn> txns,
}) {
  final b = StringBuffer()
    ..writeln('Halo $name, mohon maaf mengganggu 🙏')
    ..writeln(
        'Ini pengingat ramah dari warung: sisa utang Anda saat ini ${rupiah(balance)}.');
  if (due != null) b.writeln('Jatuh tempo: ${tgl(due)}.');
  final recent = _newestFirst(txns);
  if (recent.isNotEmpty) {
    b.writeln('\nRincian terakhir:');
    for (final t in recent) {
      b.writeln('- ${_line(t)}');
    }
  }
  b.write('\nKalau sudah ada rezekinya, ditunggu ya. Terima kasih banyak!');
  return b.toString();
}

/// Pesan bukti pelunasan untuk satu siklus utang yang telah ditutup.
String buildSettlementReceipt({
  required String name,
  required List<Txn> txns,
}) {
  final sorted = [...txns]..sort((a, b) {
    final byDate = a.date.compareTo(b.date);
    return byDate != 0 ? byDate : (a.id ?? 0).compareTo(b.id ?? 0);
  });

  final totalDebt = sorted
      .where((t) => t.type == TxnType.utang)
      .fold<int>(0, (sum, t) => sum + t.amount);
  final totalPaid = sorted
      .where((t) => t.type == TxnType.bayar)
      .fold<int>(0, (sum, t) => sum + t.amount);

  final b = StringBuffer()
    ..writeln('BUKTI PELUNASAN')
    ..writeln('Nama: $name')
    ..writeln('Waktu: ${tglJam(DateTime.now())}')
    ..writeln('Status: LUNAS')
    ..writeln('Total utang: ${rupiah(totalDebt)}')
    ..writeln('Total pembayaran: ${rupiah(totalPaid)}')
    ..writeln('')
    ..writeln('Rincian transaksi:');

  for (final t in sorted) {
    b.writeln('- ${_line(t)}');
  }

  b.writeln('');
  b.write('Terima kasih sudah menyelesaikan pembayaran 🙏');
  return b.toString();
}

/// Rekap lengkap satu pelanggan sebagai teks.
String buildRecap({
  required String name,
  required int balance,
  required List<Txn> txns,
}) {
  final b = StringBuffer()
    ..writeln('REKAP UTANG: $name')
    ..writeln('Sisa utang: ${rupiah(balance)}')
    ..writeln('Per tanggal: ${tgl(DateTime.now())}')
    ..writeln('');
  final sorted = [...txns]..sort((a, b) => a.date.compareTo(b.date));
  for (final t in sorted) {
    b.writeln(_line(t));
  }
  return b.toString().trimRight();
}

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:url_launcher/url_launcher.dart';

import '../db.dart';
import '../dialogs.dart';
import '../models.dart';
import '../status_style.dart';
import '../utils.dart';

class CustomerScreen extends StatefulWidget {
  final int customerId;
  const CustomerScreen({super.key, required this.customerId});

  @override
  State<CustomerScreen> createState() => _CustomerScreenState();
}

class _CustomerScreenState extends State<CustomerScreen> {
  Customer? _customer;
  List<Txn> _txns = [];
  bool _loading = true;

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<void> _load() async {
    final db = AppDb.instance;
    final c = await db.customer(widget.customerId);
    final t = await db.txnsFor(widget.customerId);
    if (!mounted) return;
    if (c == null) {
      Navigator.pop(context);
      return;
    }
    setState(() {
      _customer = c;
      _txns = t;
      _loading = false;
    });
  }

  void _snack(String msg) {
    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(SnackBar(content: Text(msg)));
  }

  Future<void> _addTxn(TxnType type) async {
    final c = _customer!;
    final oldBalance = computeBalance(_txns);

    final txn = await showTxnForm(
      context,
      customer: c,
      type: type,
      balance: oldBalance,
    );
    if (txn == null) return;

    await AppDb.instance.insertTxn(txn);
    await _load();

    if (!mounted) return;

    final newBalance = computeBalance(_txns);
    _snack(type == TxnType.utang ? 'Utang dicatat' : 'Pembayaran dicatat');

    if (type == TxnType.bayar && oldBalance > 0 && newBalance <= 0) {
      await _sendSettlementReceipt();
    }
  }

  Future<void> _sendSettlementReceipt() async {
    final c = _customer!;
    final cycle = latestClosedDebtCycle(_txns);
    if (cycle.isEmpty) return;

    final send = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('Utang sudah lunas'),
        content: const Text(
          'Saldo pelanggan sudah lunas. Kirim bukti pelunasan '
          'beserta seluruh transaksi dalam siklus ini melalui WhatsApp?',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx, false),
            child: const Text('Nanti'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(ctx, true),
            child: const Text('Kirim WhatsApp'),
          ),
        ],
      ),
    );

    if (send != true || !mounted) return;

    final msg = buildSettlementReceipt(name: c.name, txns: cycle);
    var opened = false;

    try {
      opened = await launchUrl(
        waUri(c.phone, msg),
        mode: LaunchMode.externalApplication,
      );
    } catch (_) {
      opened = false;
    }

    if (!opened && mounted) {
      await Clipboard.setData(ClipboardData(text: msg));
      _snack('WhatsApp tidak bisa dibuka. Bukti pelunasan disalin.');
    }
  }

  Future<void> _deleteTxn(Txn t) async {
    final ok = await confirmDialog(
      context,
      title: 'Hapus catatan?',
      message: '${t.type == TxnType.utang ? 'Utang' : 'Pembayaran'} '
          '${rupiah(t.amount)} tanggal ${tglJam(t.date)} akan dihapus.',
    );
    if (!ok) return;
    await AppDb.instance.deleteTxn(t.id!);
    await _load();
  }

  Future<void> _tagih() async {
    final c = _customer!;
    final msg = buildReminder(
      name: c.name,
      balance: computeBalance(_txns),
      due: nextDueDate(activeDebtCycle(_txns)),
      txns: activeDebtCycle(_txns),
    );
    bool ok;
    try {
      ok = await launchUrl(waUri(c.phone, msg),
          mode: LaunchMode.externalApplication);
    } catch (_) {
      ok = false;
    }
    if (!ok && mounted) {
      await Clipboard.setData(ClipboardData(text: msg));
      _snack('WhatsApp tidak bisa dibuka. Pesan disalin, tempel manual ya.');
    }
  }

  Future<void> _copyRecap() async {
    final c = _customer!;
    await Clipboard.setData(ClipboardData(
      text: buildRecap(
        name: c.name,
        balance: computeBalance(_txns),
        txns: _txns,
      ),
    ));
    if (mounted) _snack('Rekap disalin');
  }

  Future<void> _edit() async {
    final updated = await showCustomerForm(context, existing: _customer);
    if (updated == null) return;
    await AppDb.instance.updateCustomer(updated);
    await _load();
  }

  Future<void> _deleteCustomer() async {
    final c = _customer!;
    final ok = await confirmDialog(
      context,
      title: 'Hapus pelanggan?',
      message:
          '${c.name} beserta semua catatan utangnya akan dihapus permanen.',
    );
    if (!ok) return;
    await AppDb.instance.deleteCustomer(c.id!);
    if (mounted) Navigator.pop(context);
  }

  @override
  Widget build(BuildContext context) {
    if (_loading || _customer == null) {
      return Scaffold(
        appBar: AppBar(),
        body: const Center(child: CircularProgressIndicator()),
      );
    }

    final c = _customer!;
    final summary = CustomerSummary(
      customer: c,
      balance: computeBalance(_txns),
      nextDue: nextDueDate(_txns),
    );
    final history = [..._txns]..sort((a, b) {
        final byDate = b.date.compareTo(a.date);
        return byDate != 0 ? byDate : (b.id ?? 0).compareTo(a.id ?? 0);
      });

    return Scaffold(
      appBar: AppBar(
        title: Text(c.name),
        actions: [
          PopupMenuButton<String>(
            onSelected: (v) {
              switch (v) {
                case 'edit':
                  _edit();
                case 'recap':
                  _copyRecap();
                case 'delete':
                  _deleteCustomer();
              }
            },
            itemBuilder: (_) => const [
              PopupMenuItem(value: 'edit', child: Text('Ubah data')),
              PopupMenuItem(value: 'recap', child: Text('Salin rekap')),
              PopupMenuItem(value: 'delete', child: Text('Hapus pelanggan')),
            ],
          ),
        ],
      ),
      body: ListView.builder(
        padding: const EdgeInsets.only(bottom: 32),
        itemCount: history.length + 1,
        itemBuilder: (_, i) {
          if (i == 0) return _Header(summary: summary, onAdd: _addTxn, onTagih: _tagih);
          final t = history[i - 1];
          return _TxnTile(txn: t, onLongPress: () => _deleteTxn(t));
        },
      ),
    );

class _SettingsScreenState extends State<SettingsScreen> {
  bool _busy = false;
  String? _googleEmail;
  String _googleRestoreStatus = 'Belum diperiksa';
  StreamSubscription<GoogleSignInAccount?>? _googleAccountSubscription;

  @override
  void initState() {
    super.initState();

    final google = GoogleDriveService.instance;
    _googleEmail = google.account?.email;

    _googleAccountSubscription = google.accountChanges.listen(
      (account) {
        // Event tanpa akun tidak menghapus identitas yang tersimpan.
        // Logout eksplisit ditangani oleh _logoutGoogle().
        if (!mounted || account == null) return;

        setState(() {
          _googleEmail = account.email;
          _googleRestoreStatus =
              GoogleDriveService.instance.restoreDiagnostic;
        });
      },
      onError: (Object error, StackTrace stack) {
        debugPrint('Perubahan sesi Google gagal: $error');
      },
    );

    _restoreGoogleStatus();
  }

  @override
  void dispose() {
    _googleAccountSubscription?.cancel();
    super.dispose();
  }

  Future<void> _restoreGoogleStatus() async {
    try {
      final google = GoogleDriveService.instance;
      final account = await google.restoreAccount();
      final savedEmail = await google.loadSavedAccountEmail();

      if (!mounted) return;

      setState(() {
        _googleEmail = account?.email ?? savedEmail;
        final base = account != null
            ? 'SESI_AKTIF'
            : savedEmail != null
                ? 'IDENTITAS_TERSIMPAN_SESI_BELUM_AKTIF'
                : 'BELUM_TERHUBUNG';
        _googleRestoreStatus = '$base | ${google.restoreDiagnostic}';
      });
    } catch (e) {
      if (!mounted) return;
      setState(() {
        _googleRestoreStatus =
            GoogleDriveService.instance.restoreDiagnostic;
      });
      debugPrint('Pemulihan status Google gagal: $e');
    }
  }

  Future<void> _connectGoogle() async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      final email =
          await GoogleDriveService.instance.connectAndTest();

      if (!mounted) return;
      setState(() => _googleEmail = email);
      _message('Google Drive berhasil terhubung: $email');
    } catch (e) {
      _message('Koneksi Google gagal: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<void> _switchGoogle() async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      final email =
          await GoogleDriveService.instance.switchAccount();

      if (!mounted) return;
      setState(() => _googleEmail = email);
      _message('Akun Google berhasil diganti: $email');
    } catch (e) {
      _message('Ganti akun gagal: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<void> _logoutGoogle() async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      await GoogleDriveService.instance.signOut();

      if (!mounted) return;
      setState(() => _googleEmail = null);
      _message('Logout Google berhasil. Data lokal tetap aman.');
    } catch (e) {
      _message('Logout Google gagal: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  void _message(String message) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(message)),
    );
  }

  Future<void> _backup() async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      final revision = BackupStatus.instance.revision;
      final data = await AppDb.instance.exportBackup();
      final json = const JsonEncoder.withIndent('  ').convert(data);
      final now = DateTime.now();
      final stamp = '${now.year}'
          '${now.month.toString().padLeft(2, '0')}'
          '${now.day.toString().padLeft(2, '0')}_'
          '${now.hour.toString().padLeft(2, '0')}'
          '${now.minute.toString().padLeft(2, '0')}';

      final result = await FilePicker.saveFile(
        dialogTitle: 'Simpan backup Buku Utang Warung',
        fileName: 'backup_buku_utang_$stamp.json',
        bytes: Uint8List.fromList(utf8.encode(json)),
        mimeType: 'application/json',
      );

      if (result != null) {
        await BackupStatus.instance.markLocalBackedUp(revision);
        _message('File backup berhasil disimpan.');
      }
    } catch (e) {
      _message('Backup gagal: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
