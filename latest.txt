class Customer {
  final int? id;
  final String name;
  final String phone;
  final String note;

  /// Batas utang dalam rupiah. 0 berarti tanpa batas.
  final int creditLimit;
  final DateTime createdAt;

  const Customer({
    this.id,
    required this.name,
    this.phone = '',
    this.note = '',
    this.creditLimit = 0,
    required this.createdAt,
  });

  Customer copyWith({
    int? id,
    String? name,
    String? phone,
    String? note,
    int? creditLimit,
  }) {
    return Customer(
      id: id ?? this.id,
      name: name ?? this.name,
      phone: phone ?? this.phone,
      note: note ?? this.note,
      creditLimit: creditLimit ?? this.creditLimit,
      createdAt: createdAt,
    );
  }

  Map<String, Object?> toMap() => {
        if (id != null) 'id': id,
        'name': name,
        'phone': phone,
        'note': note,
        'credit_limit': creditLimit,
        'created_at': createdAt.millisecondsSinceEpoch,
      };

  factory Customer.fromMap(Map<String, Object?> m) => Customer(
        id: m['id'] as int?,
        name: m['name'] as String,
        phone: (m['phone'] as String?) ?? '',
        note: (m['note'] as String?) ?? '',
        creditLimit: (m['credit_limit'] as int?) ?? 0,
        createdAt: DateTime.fromMillisecondsSinceEpoch(m['created_at'] as int),
      );
}

/// Sisa utang = total utang dikurangi total pembayaran.
int computeBalance(List<Txn> txns) => txns.fold<int>(
      0,
      (sum, t) => sum + (t.type == TxnType.utang ? t.amount : -t.amount),
    );

/// Jatuh tempo terdekat dari utang yang belum tertutup.
/// Pembayaran dialokasikan ke utang paling lama lebih dulu (FIFO).
DateTime? nextDueDate(List<Txn> txns) {
  var credit = txns
      .where((t) => t.type == TxnType.bayar)
      .fold<int>(0, (sum, t) => sum + t.amount);
  final debts = txns.where((t) => t.type == TxnType.utang).toList()
    ..sort((a, b) => a.date.compareTo(b.date));

  DateTime? earliest;
  for (final d in debts) {
    if (credit >= d.amount) {
      credit -= d.amount;
      continue;
    }
    credit = 0;
    final due = d.dueDate;
    if (due != null && (earliest == null || due.isBefore(earliest))) {
      earliest = due;
    }
  }
  return earliest;
}

class CustomerSummary {
  final Customer customer;
  final int balance;
  final DateTime? nextDue;
  final DateTime? lastActivity;

  const CustomerSummary({
    required this.customer,
    required this.balance,
    this.nextDue,
    this.lastActivity,
  });

  DebtStatus status([DateTime? now]) {
    if (balance <= 0) return DebtStatus.lunas;
    final due = nextDue;
    if (due == null) return DebtStatus.lancar;
    final today = dateOnly(now ?? DateTime.now());
    final dueDay = dateOnly(due);
    if (dueDay.isBefore(today)) return DebtStatus.telat;
    if (dueDay.difference(today).inDays <= 3) return DebtStatus.mendekati;
    return DebtStatus.lancar;
  }

  bool get overLimit =>
      customer.creditLimit > 0 && balance > customer.creditLimit;
}

List<CustomerSummary> buildSummaries(
  List<Customer> customers,
  List<Txn> txns,
) {
  final byCustomer = <int, List<Txn>>{};
  for (final t in txns) {
    byCustomer.putIfAbsent(t.customerId, () => <Txn>[]).add(t);
  }
  return customers.map((c) {
    final list = byCustomer[c.id] ?? <Txn>[];
    return CustomerSummary(
      customer: c,
      balance: computeBalance(list),
      nextDue: nextDueDate(list),
      lastActivity: list.isEmpty
          ? null
          : list.map((t) => t.date).reduce((a, b) => a.isAfter(b) ? a : b),
    );
  }).toList();
}


/// Mengurutkan transaksi berdasarkan waktu, lalu ID untuk waktu yang sama.
List<Txn> _orderedTxns(List<Txn> txns) {
  return [...txns]..sort((a, b) {
    final byDate = a.date.compareTo(b.date);
    return byDate != 0 ? byDate : (a.id ?? 0).compareTo(b.id ?? 0);
  });
}

/// Transaksi pada siklus utang yang masih aktif.
List<Txn> activeDebtCycle(List<Txn> txns) {
  final cycle = <Txn>[];
  var balance = 0;

  for (final t in _orderedTxns(txns)) {
    cycle.add(t);
    balance += t.type == TxnType.utang ? t.amount : -t.amount;

    if (balance <= 0) {
      cycle.clear();
      balance = 0;
    }
  }

  return cycle;
}

/// Transaksi pada siklus terakhir yang sudah lunas.
List<Txn> latestClosedDebtCycle(List<Txn> txns) {
  final cycle = <Txn>[];
  List<Txn> latestClosed = [];
  var balance = 0;

  for (final t in _orderedTxns(txns)) {
    cycle.add(t);
    balance += t.type == TxnType.utang ? t.amount : -t.amount;

    if (balance <= 0) {
      if (cycle.any((item) => item.type == TxnType.utang)) {
        latestClosed = List<Txn>.from(cycle);
      }
      cycle.clear();
      balance = 0;
    }
  }

  return latestClosed;
}

const _bulan = [
  'Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun',
  'Jul', 'Agu', 'Sep', 'Okt', 'Nov', 'Des',

/// 15000 -> "Rp 15.000"
String rupiah(int value) {
  final digits = value.abs().toString();
  final buf = StringBuffer();
  for (var i = 0; i < digits.length; i++) {
    if (i > 0 && (digits.length - i) % 3 == 0) buf.write('.');
    buf.write(digits[i]);
  }
  return '${value < 0 ? '-' : ''}Rp $buf';
}

/// DateTime(2026, 11, 5) -> "5 Nov 2026"
String tglJam(DateTime d) {
  if (d.hour == 0 && d.minute == 0 && d.second == 0 && d.millisecond == 0) {
    return tgl(d);
  }
  String two(int n) => n.toString().padLeft(2, '0');
  return '${two(d.day)}-${two(d.month)}-${d.year} '
      '${two(d.hour)}:${two(d.minute)}:${two(d.second)}';
}

String tgl(DateTime d) => '${d.day} ${_bulan[d.month - 1]} ${d.year}';

String statusText(CustomerSummary s, [DateTime? now]) {
  final today = dateOnly(now ?? DateTime.now());
  switch (s.status(now)) {
    case DebtStatus.lunas:
      return 'Lunas';
    case DebtStatus.telat:
      final days = today.difference(dateOnly(s.nextDue!)).inDays;
      return 'Telat $days hari';
    case DebtStatus.mendekati:
      final days = dateOnly(s.nextDue!).difference(today).inDays;
      return days == 0 ? 'Jatuh tempo hari ini' : 'Jatuh tempo $days hari lagi';
    case DebtStatus.lancar:
      return s.nextDue == null ? 'Utang berjalan' : 'Tempo ${tgl(s.nextDue!)}';
  }
}

/// Ubah nomor lokal ke format internasional tanpa plus untuk wa.me.
/// "0812-3456-789" -> "628123456789". Mengembalikan null jika kosong.
String? normalizePhone(String raw) {
  final d = raw.replaceAll(RegExp(r'[^0-9]'), '');
  if (d.isEmpty) return null;
  if (d.startsWith('62')) return d;
  if (d.startsWith('0')) return '62${d.substring(1)}';
  if (d.startsWith('8')) return '62$d';
  return d;
}

/// Tautan WhatsApp. Tanpa nomor, WhatsApp membiarkan pengguna memilih kontak.
Uri waUri(String phone, String text) {
  final n = normalizePhone(phone);
  final encoded = Uri.encodeComponent(text);
  return Uri.parse('https://wa.me/${n ?? ''}?text=$encoded');
}

String _line(Txn t) {
  final sign = t.type == TxnType.utang ? '+' : '-';
  final label = t.type == TxnType.utang ? 'utang' : 'bayar';
  final note = t.note.trim().isEmpty ? '' : ' (${t.note.trim()})';
  return '${tglJam(t.date)}: $label $sign${rupiah(t.amount)}$note';
}

List<Txn> _newestFirst(List<Txn> txns) =>
    [...txns]..sort((a, b) => b.date.compareTo(a.date));

/// Pesan tagih yang sopan, siap kirim lewat WhatsApp.
String buildReminder({
  required String name,
  required int balance,
  DateTime? due,
  required List<Txn> txns,
}) {
  final b = StringBuffer()
    ..writeln('Halo $name, mohon maaf mengganggu 🙏')
    ..writeln(
        'Ini pengingat ramah dari warung: sisa utang Anda saat ini ${rupiah(balance)}.');
  if (due != null) b.writeln('Jatuh tempo: ${tgl(due)}.');
  final recent = _newestFirst(txns);
  if (recent.isNotEmpty) {
    b.writeln('\nRincian terakhir:');
    for (final t in recent) {
      b.writeln('- ${_line(t)}');
    }
  }
  b.write('\nKalau sudah ada rezekinya, ditunggu ya. Terima kasih banyak!');
  return b.toString();
}

/// Pesan bukti pelunasan untuk satu siklus utang yang telah ditutup.
String buildSettlementReceipt({
  required String name,
  required List<Txn> txns,
}) {
  final sorted = [...txns]..sort((a, b) {
    final byDate = a.date.compareTo(b.date);
    return byDate != 0 ? byDate : (a.id ?? 0).compareTo(b.id ?? 0);
  });

  final totalDebt = sorted
      .where((t) => t.type == TxnType.utang)
      .fold<int>(0, (sum, t) => sum + t.amount);
  final totalPaid = sorted
      .where((t) => t.type == TxnType.bayar)
      .fold<int>(0, (sum, t) => sum + t.amount);

  final b = StringBuffer()
    ..writeln('BUKTI PELUNASAN')
    ..writeln('Nama: $name')
    ..writeln('Waktu: ${tglJam(DateTime.now())}')
    ..writeln('Status: LUNAS')
    ..writeln('Total utang: ${rupiah(totalDebt)}')
    ..writeln('Total pembayaran: ${rupiah(totalPaid)}')
    ..writeln('')
    ..writeln('Rincian transaksi:');

  for (final t in sorted) {
    b.writeln('- ${_line(t)}');
  }

  b.writeln('');
  b.write('Terima kasih sudah menyelesaikan pembayaran 🙏');
  return b.toString();
}

/// Rekap lengkap satu pelanggan sebagai teks.
String buildRecap({

  required List<Txn> txns,
}) {
  final b = StringBuffer()
    ..writeln('REKAP UTANG: $name')
    ..writeln('Sisa utang: ${rupiah(balance)}')
    ..writeln('Per tanggal: ${tgl(DateTime.now())}')
    ..writeln('');
  final sorted = [...txns]..sort((a, b) => a.date.compareTo(b.date));
  for (final t in sorted) {
    b.writeln(_line(t));
  }
  return b.toString().trimRight();
}

  State<_TxnDialog> createState() => _TxnDialogState();
}

class _TxnDialogState extends State<_TxnDialog> {
  final _amount = TextEditingController();
  final _note = TextEditingController();
  String? _error;

  // 0 = tanpa tempo, 1 = 1 minggu, 2 = 2 minggu, 3 = 1 bulan, 4 = pilih tanggal
  int _dueChoice = 0;
  DateTime? _due;

  bool get _isDebt => widget.type == TxnType.utang;
  int get _value => int.tryParse(_amount.text) ?? 0;

  @override
  void dispose() {
    _amount.dispose();
    _note.dispose();
    super.dispose();
  }

  void _pickDue(int choice) {
    final today = dateOnly(DateTime.now());
    setState(() {
      _dueChoice = choice;
      switch (choice) {
        case 1:
          _due = today.add(const Duration(days: 7));
          break;
        case 2:
          _due = today.add(const Duration(days: 14));
          break;
        case 3:
          _due = DateTime(today.year, today.month + 1, today.day);
          break;
        default:
          _due = null;
      }
    });
  }

  Future<void> _pickCustomDate() async {
    final today = dateOnly(DateTime.now());
    final picked = await showDatePicker(
      context: context,
      initialDate: _due ?? today.add(const Duration(days: 7)),
      firstDate: today,
      lastDate: today.add(const Duration(days: 365 * 3)),
    );
    if (picked == null) return;
    setState(() {
      _dueChoice = 4;
      _due = dateOnly(picked);
    });
  }

  void _save() {
    if (_value <= 0) {
      setState(() => _error = 'Isi nominal dulu');
      return;
    }
    Navigator.pop(
      context,
      Txn(
        customerId: widget.customer.id!,
        type: widget.type,
        amount: _value,
        note: _note.text.trim(),
        date: DateTime.now(),
        dueDate: _isDebt ? _due : null,
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final limit = widget.customer.creditLimit;
    final overLimit =
        _isDebt && limit > 0 && widget.balance + _value > limit;

    return AlertDialog(
      title: Text(_isDebt ? 'Utang baru' : 'Pembayaran'),
      content: SingleChildScrollView(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(widget.customer.name,
                style: Theme.of(context).textTheme.titleMedium),
            if (widget.balance != 0)
              Text('Sisa utang sekarang: ${rupiah(widget.balance)}',
                  style: Theme.of(context).textTheme.bodySmall),
            const SizedBox(height: 8),
            TextField(
              controller: _amount,
              autofocus: true,
              keyboardType: TextInputType.number,
              inputFormatters: [FilteringTextInputFormatter.digitsOnly],
              style: Theme.of(context).textTheme.headlineSmall,
              decoration: InputDecoration(
                labelText: 'Nominal',
                prefixText: 'Rp ',
                errorText: _error,
              ),
              onChanged: (_) => setState(() => _error = null),
            ),
            if (_value > 0)
              Padding(
                padding: const EdgeInsets.only(top: 4),
                child: Text(rupiah(_value),
                    style: const TextStyle(fontWeight: FontWeight.bold)),
              ),
            if (!_isDebt && widget.balance > 0)
              Align(
                alignment: Alignment.centerLeft,
                child: TextButton(
                  onPressed: () => setState(() {
                    _amount.text = '${widget.balance}';
                    _error = null;
                  }),
                  child: Text('Lunas (${rupiah(widget.balance)})'),
                ),
              ),
            if (overLimit)
              Padding(
                padding: const EdgeInsets.only(top: 4),
                child: Text(
                  'Melewati batas utang ${rupiah(limit)}',
                  style: const TextStyle(
                      color: kMerah, fontWeight: FontWeight.w600),
                ),
              ),
            TextField(
              controller: _note,
              textCapitalization: TextCapitalization.sentences,
              decoration: InputDecoration(
                labelText: 'Catatan (opsional)',
                hintText: _isDebt ? 'Contoh: beras 5kg, minyak' : 'Contoh: cicil',
              ),
            ),
            if (_isDebt) ...[
              const SizedBox(height: 16),
              Text('Jatuh tempo', style: Theme.of(context).textTheme.labelLarge),
              const SizedBox(height: 4),
              Wrap(
                spacing: 8,
                children: [
                  ChoiceChip(
                    label: const Text('Tanpa'),
                    selected: _dueChoice == 0,
                    onSelected: (_) => _pickDue(0),
                  ),
                  ChoiceChip(
                    label: const Text('1 minggu'),
                    selected: _dueChoice == 1,
                    onSelected: (_) => _pickDue(1),
                  ),
                  ChoiceChip(
                    label: const Text('2 minggu'),
                    selected: _dueChoice == 2,
                    onSelected: (_) => _pickDue(2),
                  ),
                  ChoiceChip(
                    label: const Text('1 bulan'),
                    selected: _dueChoice == 3,
                    onSelected: (_) => _pickDue(3),
                  ),
                  ChoiceChip(
                    label: Text(_dueChoice == 4 && _due != null
                        ? tgl(_due!)
                        : 'Pilih tanggal'),
                    selected: _dueChoice == 4,
                    onSelected: (_) => _pickCustomDate(),
                  ),
                ],
              ),
              if (_due != null && _dueChoice != 4)
                Padding(
                  padding: const EdgeInsets.only(top: 6),
                  child: Text('Jatuh tempo: ${tgl(_due!)}',
                      style: Theme.of(context).textTheme.bodySmall),
                ),
            ],
          ],
        ),
      ),
      actions: [
        TextButton(
            onPressed: () => Navigator.pop(context),
            child: const Text('Batal')),
        FilledButton(onPressed: _save, child: const Text('Simpan')),
      ],
    );
  }

    final oldBalance = computeBalance(_txns);

    final txn = await showTxnForm(
      context,
      customer: c,
      type: type,
      balance: oldBalance,
    );
    if (txn == null) return;

    final newBalance = computeBalance(_txns);
    _snack(type == TxnType.utang ? 'Utang dicatat' : 'Pembayaran dicatat');

            ],
          ),
        ],
      ),
      body: ListView.builder(
        padding: const EdgeInsets.only(bottom: 32),
        itemCount: history.length + 1,
        itemBuilder: (_, i) {
          if (i == 0) return _Header(summary: summary, onAdd: _addTxn, onTagih: _tagih);
          final t = history[i - 1];
          return _TxnTile(txn: t, onLongPress: () => _deleteTxn(t));
        },
      ),
    );
  }
}

class _TxnTile extends StatelessWidget {
  final Txn txn;
  final VoidCallback onLongPress;
  const _TxnTile({required this.txn, required this.onLongPress});

  @override
  Widget build(BuildContext context) {
    final isDebt = txn.type == TxnType.utang;
    final color = isDebt ? kMerah : kHijau;
    final subtitle = [
      tglJam(txn.date),
      if (isDebt && txn.dueDate != null) 'tempo ${tgl(txn.dueDate!)}',
    ].join(' • ');

    return ListTile(
      onLongPress: onLongPress,
      leading: CircleAvatar(
        backgroundColor: color.withAlpha(35),
        foregroundColor: color,
        child: Icon(isDebt ? Icons.arrow_upward : Icons.arrow_downward),
      ),
      title: Text(txn.note.isEmpty
          ? (isDebt ? 'Utang' : 'Pembayaran')
          : txn.note),
      subtitle: Text(subtitle),
      trailing: Text(
        '${isDebt ? '+' : '-'}${rupiah(txn.amount)}',
        style: TextStyle(
            color: color, fontWeight: FontWeight.bold, fontSize: 15),
      ),
    );
  }
}
