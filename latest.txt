===== WORKFLOW: SCAFFOLD ANDROID =====
name: Build APK

on:
  push:
    branches: [main, master]
  workflow_dispatch:

permissions:
  contents: read

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v5

      - name: Setup Java
        uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: "21"

      - name: Setup Flutter
        uses: subosito/flutter-action@v2
        with:
          channel: stable
          cache: true

      # Buat scaffold Android tanpa kehilangan kode aplikasi
      - name: Buat scaffold Android
        run: |
          set -euo pipefail

          mkdir -p "$RUNNER_TEMP/bak"
          cp -r lib test pubspec.yaml "$RUNNER_TEMP/bak/"

          flutter create --platforms=android \
            --org id.warung \
            --project-name buku_utang_warung .

          rm -rf lib test pubspec.yaml
          cp -r "$RUNNER_TEMP/bak/lib" lib
          cp -r "$RUNNER_TEMP/bak/test" test
          cp "$RUNNER_TEMP/bak/pubspec.yaml" pubspec.yaml

          sed -i 's/android:label="[^"]*"/android:label="Buku Utang Warung"/' \
            android/app/src/main/AndroidManifest.xml

          # Kunci Android Keystore tidak boleh dipindahkan oleh Android Auto Backup.
          python3 - <<'PYSECURE'
          from pathlib import Path

          manifest = Path("android/app/src/main/AndroidManifest.xml")
          text = manifest.read_text()
          start = text.find("<application")
          if start < 0:
              raise SystemExit("Tag application tidak ditemukan.")

          end = text.find(">", start)
          if end < 0:
              raise SystemExit("Tag application tidak lengkap.")

          tag = text[start:end + 1]
          if "android:allowBackup=" in tag:
              raise SystemExit("allowBackup sudah diatur; periksa manual.")

          text = text[:end] + ' android:allowBackup="false"' + text[end:]
          manifest.write_text(text)
          print("Android Auto Backup dinonaktifkan.")
          PYSECURE


      # Siapkan keystore dari GitHub Secrets
