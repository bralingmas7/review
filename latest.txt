   100	            : DateTime.fromMillisecondsSinceEpoch(m['due_date'] as int),
   101	      );
   102	}
   103	
   104	/// Sisa utang = total utang dikurangi total pembayaran.
   105	int computeBalance(List<Txn> txns) => txns.fold<int>(
   106	      0,
   107	      (sum, t) => sum + (t.type == TxnType.utang ? t.amount : -t.amount),
   108	    );
   109	
   110	/// Jatuh tempo terdekat dari utang yang belum tertutup.
   111	/// Pembayaran dialokasikan ke utang paling lama lebih dulu (FIFO).
   112	DateTime? nextDueDate(List<Txn> txns) {
   113	  var credit = txns
   114	      .where((t) => t.type == TxnType.bayar)
   115	      .fold<int>(0, (sum, t) => sum + t.amount);
   116	  final debts = txns.where((t) => t.type == TxnType.utang).toList()
   117	    ..sort((a, b) => a.date.compareTo(b.date));
   118	
   119	  DateTime? earliest;
   120	  for (final d in debts) {
   121	    if (credit >= d.amount) {
   122	      credit -= d.amount;
   123	      continue;
   124	    }
   125	    credit = 0;
   126	    final due = d.dueDate;
   127	    if (due != null && (earliest == null || due.isBefore(earliest))) {
   128	      earliest = due;
   129	    }
   130	  }
   131	  return earliest;
   132	}

   134	class CustomerSummary {
   135	  final Customer customer;
   136	  final int balance;
   137	  final DateTime? nextDue;
   138	  final DateTime? lastActivity;
   139	
   140	  const CustomerSummary({
   141	    required this.customer,
   142	    required this.balance,
   143	    this.nextDue,
   144	    this.lastActivity,
   145	  });
   146	
   147	  DebtStatus status([DateTime? now]) {
   148	    if (balance <= 0) return DebtStatus.lunas;
   149	    final due = nextDue;
   150	    if (due == null) return DebtStatus.lancar;
   151	    final today = dateOnly(now ?? DateTime.now());
   152	    final dueDay = dateOnly(due);
   153	    if (dueDay.isBefore(today)) return DebtStatus.telat;
   154	    if (dueDay.difference(today).inDays <= 3) return DebtStatus.mendekati;
   155	    return DebtStatus.lancar;
   156	  }
   157	
   158	  bool get overLimit =>
   159	      customer.creditLimit > 0 && balance > customer.creditLimit;
   160	}
   161	
   162	List<CustomerSummary> buildSummaries(
   163	  List<Customer> customers,
   164	  List<Txn> txns,
   165	) {
   166	  final byCustomer = <int, List<Txn>>{};
   167	  for (final t in txns) {
   168	    byCustomer.putIfAbsent(t.customerId, () => <Txn>[]).add(t);
   169	  }
   170	  return customers.map((c) {
   171	    final list = byCustomer[c.id] ?? <Txn>[];
   172	    return CustomerSummary(
   173	      customer: c,
   174	      balance: computeBalance(list),
   175	      nextDue: nextDueDate(list),
   176	      lastActivity: list.isEmpty
   177	          ? null
   178	          : list.map((t) => t.date).reduce((a, b) => a.isAfter(b) ? a : b),
   179	    );
   180	  }).toList();
   181	}

   184	/// Mengurutkan transaksi berdasarkan waktu, lalu ID untuk waktu yang sama.
   185	List<Txn> _orderedTxns(List<Txn> txns) {
   186	  return [...txns]..sort((a, b) {
   187	    final byDate = a.date.compareTo(b.date);
   188	    return byDate != 0 ? byDate : (a.id ?? 0).compareTo(b.id ?? 0);
   189	  });
   190	}

   192	/// Transaksi pada siklus utang yang masih aktif.
   193	List<Txn> activeDebtCycle(List<Txn> txns) {
   194	  final cycle = <Txn>[];
   195	  var balance = 0;
   196	
   197	  for (final t in _orderedTxns(txns)) {
   198	    cycle.add(t);
   199	    balance += t.type == TxnType.utang ? t.amount : -t.amount;
   200	
   201	    if (balance <= 0) {
   202	      cycle.clear();
   203	      balance = 0;
   204	    }
   205	  }
   206	
   207	  return cycle;
   208	}
   209	
   210	/// Transaksi pada siklus terakhir yang sudah lunas.
   211	List<Txn> latestClosedDebtCycle(List<Txn> txns) {
   212	  final cycle = <Txn>[];
   213	  List<Txn> latestClosed = [];
   214	  var balance = 0;
   215	
   216	  for (final t in _orderedTxns(txns)) {
   217	    cycle.add(t);
   218	    balance += t.type == TxnType.utang ? t.amount : -t.amount;
   219	
   220	    if (balance <= 0) {
   221	      if (cycle.any((item) => item.type == TxnType.utang)) {
   222	        latestClosed = List<Txn>.from(cycle);
   223	      }
   224	      cycle.clear();
   225	      balance = 0;
   226	    }
   227	  }
   228	
   229	  return latestClosed;
   230	}

    41	      // anggap belum ter-backup. Kalau masih kosong, anggap aman.
    42	      driveDirty.value ??= drive ?? hasData;
    43	      localDirty.value ??= local ?? hasData;
    44	
    45	      if (drive == null) await _persist(_driveKey, driveDirty.value ?? hasData);
    46	      if (local == null) await _persist(_localKey, localDirty.value ?? hasData);
    47	    } catch (e) {
    48	      debugPrint('Gagal memuat status backup: $e');
    49	      driveDirty.value ??= hasData;
    50	      localDirty.value ??= hasData;
    51	    }
    52	  }
    53	
    54	  void markDirty() {
    55	    _revision++;
    56	    driveDirty.value = true;
    57	    localDirty.value = true;
    58	    unawaited(_persist(_driveKey, true));
    59	    unawaited(_persist(_localKey, true));
    60	  }
    61	
    62	  /// Tandai Google Drive sudah ter-backup, hanya jika data tidak berubah
    63	  /// sejak [revision] dicatat (sebelum backup dibuat).
    64	  Future<void> markDriveBackedUp(int revision) async {
    65	    if (revision != _revision) return;
    66	    driveDirty.value = false;
    67	    await _persist(_driveKey, false);
    68	  }
    69	
    70	  /// Sama seperti [markDriveBackedUp], untuk backup file lokal.
    71	  Future<void> markLocalBackedUp(int revision) async {
    72	    if (revision != _revision) return;
    73	    localDirty.value = false;
    74	    await _persist(_localKey, false);
    75	  }
    76	
    77	  /// Nama lama, tetap dipakai oleh alur backup Google Drive.
    78	  Future<void> markBackedUp(int revision) => markDriveBackedUp(revision);
    79	
    80	  Future<void> _persist(String key, bool value) async {
    81	    try {
    82	      await SharedPreferencesAsync().setBool(key, value);
    83	    } catch (e) {
    84	      debugPrint('Gagal menyimpan status backup: $e');
    85	    }
    86	  }
    87	}
