     6	/// Melacak apakah ada perubahan data yang belum di-backup, terpisah untuk
     7	/// backup lokal (file di HP) dan Google Drive.
     8	///
     9	/// Nilai notifier: null = belum diketahui, true = ada perubahan belum
    10	/// di-backup, false = sudah di-backup.
    11	class BackupStatus {
    12	  BackupStatus._();
    13	
    14	  static final BackupStatus instance = BackupStatus._();
    15	
    16	  // Kunci lama dipertahankan untuk Google Drive agar status lama tidak hilang.
    17	  static const String _driveKey = 'backup_dirty';
    18	  static const String _localKey = 'backup_local_dirty';
    19	
    20	  final ValueNotifier<bool?> driveDirty = ValueNotifier<bool?>(null);
    21	  final ValueNotifier<bool?> localDirty = ValueNotifier<bool?>(null);
    22	
    23	  bool _loaded = false;
    24	  int _revision = 0;
    25	
    26	  // Semua penulisan status dijalankan berurutan agar status lama
    27	  // tidak menimpa status terbaru ketika operasi berlangsung bersamaan.
    28	  Future<void> _writeQueue = Future<void>.value();
    29	
    30	  /// Naik setiap data berubah. Dipakai agar perubahan yang terjadi saat
    31	  /// backup berlangsung tidak ikut dianggap sudah ter-backup.
    32	  int get revision => _revision;
    33	
    34	  /// Dipanggil sekali saat app dibuka. Panggilan berikutnya diabaikan.
    35	  Future<void> load({required bool hasData}) async {
    36	    if (_loaded) return;
    37	    _loaded = true;
    38	
    39	    try {
    40	      final prefs = SharedPreferencesAsync();
    41	      final drive = await prefs.getBool(_driveKey);
    42	      final local = await prefs.getBool(_localKey);
    43	
    44	      // Pertama kali: kalau sudah ada data tapi belum pernah di-backup,
    45	      // anggap belum ter-backup. Kalau masih kosong, anggap aman.
    46	      driveDirty.value ??= drive ?? hasData;
    47	      localDirty.value ??= local ?? hasData;
    48	
    49	      if (drive == null) await _persist(_driveKey, driveDirty.value ?? hasData);
    50	      if (local == null) await _persist(_localKey, localDirty.value ?? hasData);
    51	    } catch (e) {
    52	      debugPrint('Gagal memuat status backup: $e');
    53	      driveDirty.value ??= hasData;
    54	      localDirty.value ??= hasData;
    55	    }
    56	  }
    57	
    58	  void markDirty() {
    59	    _revision++;
    60	    driveDirty.value = true;
    61	    localDirty.value = true;
    62	    unawaited(_persist(_driveKey, true));
    63	    unawaited(_persist(_localKey, true));
    64	  }
    65	
    66	  /// Tandai Google Drive sudah ter-backup, hanya jika data tidak berubah
    67	  /// sejak [revision] dicatat (sebelum backup dibuat).
    68	  Future<void> markDriveBackedUp(int revision) async {
    69	    if (revision != _revision) return;
    70	    driveDirty.value = false;
    71	    await _persist(_driveKey, false);
    72	  }
    73	
    74	  /// Sama seperti [markDriveBackedUp], untuk backup file lokal.
    75	  Future<void> markLocalBackedUp(int revision) async {
    76	    if (revision != _revision) return;
    77	    localDirty.value = false;
    78	    await _persist(_localKey, false);
    79	  }
    80	
    81	  /// Nama lama, tetap dipakai oleh alur backup Google Drive.
    82	  Future<void> markBackedUp(int revision) => markDriveBackedUp(revision);
    83	
    84	  Future<void> _persist(String key, bool value) {
    85	    final operation = _writeQueue.then((_) async {
    86	      try {
    87	        await SharedPreferencesAsync().setBool(key, value);
    88	      } catch (e) {
    89	        debugPrint('Gagal menyimpan status backup: $e');
    90	      }
    91	    });
    92	
    93	    _writeQueue = operation;
    94	    return operation;
    95	  }

   136	    Object? value,
   137	    String label,
   138	  ) {
   139	    if (value is! List) {
   140	      throw FormatException('Bagian $label bukan daftar yang valid');
   141	    }
   142	
   143	    return value.map((item) {
   144	      if (item is! Map) {
   145	        throw FormatException('Ada data $label yang tidak valid');
   146	      }
   147	      return Map<String, Object?>.from(item);
   148	    }).toList();

   151	  // Validasi lengkap dilakukan sebelum database boleh diubah.
   152	  static void validateBackupData(Map<String, dynamic> data) {
   153	    if (data['format'] != backupFormat) {
   154	      throw const FormatException('Format file bukan backup Buku Utang Warung');
   155	    }
   156	    if (data['version'] != backupVersion) {
   157	      throw const FormatException('Versi backup tidak didukung');
   158	    }
   159	
   160	    final customers = _rows(data['customers'], 'pelanggan');
   161	    final transactions = _rows(data['transactions'], 'transaksi');
   162	
   163	    final customerIds = <int>{};
   164	
   165	    for (final row in customers) {
   166	      final id = row['id'];
   167	      if (id is! int || id <= 0 || !customerIds.add(id)) {
   168	        throw const FormatException('ID pelanggan tidak valid atau duplikat');
   169	      }
   170	      if (row['name'] is! String ||
   171	          row['phone'] is! String ||
   172	          row['note'] is! String ||
   173	          row['credit_limit'] is! int ||
   174	          (row['credit_limit'] as int) < 0 ||
   175	          row['created_at'] is! int) {
   176	        throw const FormatException('Data pelanggan tidak lengkap atau rusak');
   177	      }
   178	    }
   179	
   180	    final txnIds = <int>{};
   181	
   182	    for (final row in transactions) {
   183	      final id = row['id'];
   184	      final customerId = row['customer_id'];
   185	      final type = row['type'];
   186	      final amount = row['amount'];
   187	      final dueDate = row['due_date'];
   188	
   189	      if (id is! int || id <= 0 || !txnIds.add(id)) {
   190	        throw const FormatException('ID transaksi tidak valid atau duplikat');
   191	      }
   192	      if (customerId is! int || !customerIds.contains(customerId)) {
   193	        throw const FormatException(
   194	          'Transaksi mengacu ke pelanggan yang tidak ditemukan',
   195	        );
   196	      }
   197	      if (type != 'utang' && type != 'bayar') {
   198	        throw const FormatException('Jenis transaksi tidak dikenal');
   199	      }
   200	      if (amount is! int || amount <= 0 ||
   201	          row['note'] is! String ||
   202	          row['date'] is! int ||
   203	          (dueDate != null && dueDate is! int)) {
   204	        throw const FormatException('Data transaksi tidak lengkap atau rusak');
   205	      }
   206	    }
   207	  }
   208	
   209	  // Penggantian seluruh data dilakukan dalam satu transaksi SQLite.
   210	  // Jika satu insert gagal, SQLite membatalkan seluruh proses.
   211	  Future<void> restoreBackup(Map<String, dynamic> data) async {
   212	    validateBackupData(data);
   213	
   214	    final customers = _rows(data['customers'], 'pelanggan');
   215	    final transactions = _rows(data['transactions'], 'transaksi');
   216	    final db = await _database;
   217	
   218	    await db.transaction((txn) async {
   219	      await txn.delete('transactions');
   220	      await txn.delete('customers');
   221	
   222	      for (final row in customers) {
   223	        await txn.insert('customers', row);
   224	      }
   225	
   226	      for (final row in transactions) {
   227	        await txn.insert('transactions', row);
   228	      }
   229	    });
   230	
   231	    BackupStatus.instance.markDirty();
   232	  }
