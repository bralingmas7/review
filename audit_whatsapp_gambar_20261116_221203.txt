     9	String rupiah(int value) {
    10	  final digits = value.abs().toString();
    11	  final buf = StringBuffer();
    12	  for (var i = 0; i < digits.length; i++) {
    13	    if (i > 0 && (digits.length - i) % 3 == 0) buf.write('.');
    14	    buf.write(digits[i]);
    15	  }
    16	  return '${value < 0 ? '-' : ''}Rp $buf';
    17	}

    20	String tglJam(DateTime d) {
    21	  if (d.hour == 0 && d.minute == 0 && d.second == 0 && d.millisecond == 0) {
    22	    return tgl(d);
    23	  }
    24	  String two(int n) => n.toString().padLeft(2, '0');
    25	  return '${two(d.day)}-${two(d.month)}-${d.year} '
    26	      '${two(d.hour)}:${two(d.minute)}:${two(d.second)}';
    27	}

    32	  final today = dateOnly(now ?? DateTime.now());
    33	  switch (s.status(now)) {
    34	    case DebtStatus.lunas:

    50	  final d = raw.replaceAll(RegExp(r'[^0-9]'), '');

    54	  if (d.startsWith('8')) return '62$d';

   110	/// Jatuh tempo terdekat dari utang yang belum tertutup.
   111	/// Pembayaran dialokasikan ke utang paling lama lebih dulu (FIFO).
   112	DateTime? nextDueDate(List<Txn> txns) {
   113	  var credit = txns
   114	      .where((t) => t.type == TxnType.bayar)
   115	      .fold<int>(0, (sum, t) => sum + t.amount);
   116	  final debts = txns.where((t) => t.type == TxnType.utang).toList()
   117	    ..sort((a, b) => a.date.compareTo(b.date));
   118	
   119	  DateTime? earliest;
   120	  for (final d in debts) {
   121	    if (credit >= d.amount) {
   122	      credit -= d.amount;
   123	      continue;
   124	    }
   125	    credit = 0;
   126	    final due = d.dueDate;
   127	    if (due != null && (earliest == null || due.isBefore(earliest))) {
   128	      earliest = due;
   129	    }
   130	  }
   131	  return earliest;
   132	}

   147	  DebtStatus status([DateTime? now]) {

   193	List<Txn> activeDebtCycle(List<Txn> txns) {
   194	  final cycle = <Txn>[];
   195	  var balance = 0;
   196	
   197	  for (final t in _orderedTxns(txns)) {
   198	    cycle.add(t);
   199	    balance += t.type == TxnType.utang ? t.amount : -t.amount;
   200	
   201	    if (balance <= 0) {

   203	      balance = 0;
   204	    }

   210	/// Transaksi pada siklus terakhir yang sudah lunas.
   211	List<Txn> latestClosedDebtCycle(List<Txn> txns) {
   212	  final cycle = <Txn>[];
   213	  List<Txn> latestClosed = [];
   214	  var balance = 0;
   215	
   216	  for (final t in _orderedTxns(txns)) {
   217	    cycle.add(t);
   218	    balance += t.type == TxnType.utang ? t.amount : -t.amount;
   219	
   220	    if (balance <= 0) {
   221	      if (cycle.any((item) => item.type == TxnType.utang)) {
   222	        latestClosed = List<Txn>.from(cycle);
   223	      }
   224	      cycle.clear();
   225	      balance = 0;
   226	    }
   227	  }
