import 'package:flutter_secure_storage/flutter_secure_storage.dart';

class BackupCredentialService {
  BackupCredentialService._();

  static final BackupCredentialService instance =
      BackupCredentialService._();

  static const String _passwordKey = 'buw_cloud_backup_password_v1';

  // Default Android implementation uses Android Keystore to protect the
  // encryption key and AES-GCM to encrypt the stored value.
  final FlutterSecureStorage _storage = FlutterSecureStorage();

  Future<String?> readPassword() {
    return _storage.read(key: _passwordKey);
  }

  // Accept 8+ characters here to preserve access to older backups.
  // New passwords are enforced at 12+ characters by BackupCryptoService.
  Future<void> savePassword(String password) async {
    if (password.runes.length < 8) {
      throw ArgumentError('Password backup minimal 8 karakter.');
    }
    await _storage.write(key: _passwordKey, value: password);
  }

  Future<void> clearPassword() {
    return _storage.delete(key: _passwordKey);
  }
}

class BackupCryptoService {
  BackupCryptoService._();

  static final BackupCryptoService instance = BackupCryptoService._();

  static const int minimumPasswordLength = 12;
  static const int _legacyMinimumPasswordLength = 8;
  static const int _version = 1;
  static const int _iterations = 310000;
  static const int _maximumEncryptedBytes = 100 * 1024 * 1024;

  static const String _format = 'buku_utang_warung_encrypted_backup';
  static const String _cipherName = 'AES-256-GCM';
  static const String _kdfName = 'PBKDF2-HMAC-SHA256';

  final Pbkdf2 _kdf = Pbkdf2(
    macAlgorithm: Hmac.sha256(),
    iterations: _iterations,
    bits: 256,
  );

  final AesGcm _cipher = AesGcm.with256bits();

  void _validatePassword(
    String password, {
    int minimumLength = minimumPasswordLength,
  }) {
    if (password.runes.length < minimumLength) {
      throw ArgumentError(
        'Password backup minimal $minimumLength karakter.',
      );
    }
  }

  Future<Uint8List> encryptBackup(
    Map<String, Object?> backup,
    String password,
  ) async {
    _validatePassword(password);

    final salt = Uint8List(16);
    final random = Random.secure();

    for (var i = 0; i < salt.length; i++) {
      salt[i] = random.nextInt(256);
    }

    final key = await _kdf.deriveKeyFromPassword(
      password: password,
      nonce: salt,
    );

    final clearBytes = utf8.encode(jsonEncode(backup));
    final encrypted = await _cipher.encrypt(
      clearBytes,
      secretKey: key,
    );

    final envelope = <String, Object?>{
      'format': _format,
      'version': _version,
      'cipher': _cipherName,
      'kdf': _kdfName,
      'iterations': _iterations,
      'salt': base64Encode(salt),
      'nonce': base64Encode(encrypted.nonce),
      'ciphertext': base64Encode(encrypted.cipherText),
      'mac': base64Encode(encrypted.mac.bytes),
    };

    return Uint8List.fromList(utf8.encode(jsonEncode(envelope)));
  }

  Future<Map<String, dynamic>> decryptBackup(
    List<int> encryptedBytes,
    String password,
  ) async {
    // Backups created by older app versions allowed 8-character passwords.
    _validatePassword(
      password,
      minimumLength: _legacyMinimumPasswordLength,
    );

    if (encryptedBytes.isEmpty ||
        encryptedBytes.length > _maximumEncryptedBytes) {
      throw const FormatException('Ukuran file backup tidak valid.');
    }

    final decodedEnvelope = jsonDecode(utf8.decode(encryptedBytes));

    if (decodedEnvelope is! Map<String, dynamic>) {
      throw const FormatException('Format file backup tidak valid.');
    }

    final envelope = decodedEnvelope;

    if (envelope['format'] != _format ||
        envelope['version'] != _version ||
        envelope['cipher'] != _cipherName ||
        envelope['kdf'] != _kdfName ||
        envelope['iterations'] != _iterations) {
      throw const FormatException(
        'Format atau versi enkripsi backup tidak didukung.',
      );
    }

    try {
      final salt = base64Decode(envelope['salt'] as String);
      final nonce = base64Decode(envelope['nonce'] as String);
      final ciphertext = base64Decode(envelope['ciphertext'] as String);
      final macBytes = base64Decode(envelope['mac'] as String);

      if (salt.length != 16 ||
          nonce.length != 12 ||
          macBytes.length != 16 ||
          ciphertext.isEmpty) {
        throw const FormatException('Data enkripsi backup tidak lengkap.');
      }

      final key = await _kdf.deriveKeyFromPassword(
        password: password,
        nonce: salt,
      );

      final secretBox = SecretBox(
        ciphertext,
        nonce: nonce,
        mac: Mac(macBytes),
      );

      final clearBytes = await _cipher.decrypt(
        secretBox,
        secretKey: key,
      );

      final decodedBackup = jsonDecode(utf8.decode(clearBytes));

      if (decodedBackup is! Map<String, dynamic>) {
        throw const FormatException('Isi backup bukan objek JSON.');
      }

      return decodedBackup;
    } on FormatException {
      rethrow;
    } catch (_) {
      throw const FormatException(
        'Backup gagal dibuka. Password salah atau file telah berubah.',
      );
    }
  }
}

  Future<T> _withDriveApi<T>(
    Future<T> Function(drive.DriveApi api) action,
  ) async {

    final authorization =
        await authorizationClient.authorizationForScopes(_scopes) ??
        await authorizationClient.authorizeScopes(_scopes);

    final client = authorization.authClient(scopes: _scopes);

      return await action(api);

  Future<List<drive.File>> _listAllEncryptedBackups(
    drive.DriveApi api,
  ) async {
    final results = <drive.File>[];

    do {
      final response = await api.files.list(

            "appProperties has { key='app' and "

        pageSize: 100,
        pageToken: pageToken,

      results.addAll(response.files ?? <drive.File>[]);

  Future<drive.File> uploadEncryptedBackup(
    List<int> encryptedBytes,
  ) async {
    _validateEncryptedFile(encryptedBytes);

    return _withDriveApi((api) async {
      final metadata = drive.File()
        ..name = 'backup_buku_utang_warung.buwbackup'
        ..mimeType = 'application/octet-stream'
        ..appProperties = {
          'app': 'buku_utang_warung',
          'format': 'encrypted_backup',
          'version': '1',
        };

      final media = drive.Media(
        Stream<List<int>>.value(encryptedBytes),
        encryptedBytes.length,
      );

      final existing = await _listAllEncryptedBackups(api);
      late final drive.File uploaded;

      if (existing.isEmpty) {
        uploaded = await api.files.create(
          metadata,
          uploadMedia: media,
          $fields: 'id,name,mimeType,size,createdTime,modifiedTime',
        );
      } else {
        final canonical = existing.first;
        final canonicalId = canonical.id;
        if (canonicalId == null || canonicalId.isEmpty) {
          throw StateError('ID backup cloud utama tidak tersedia.');
        }

        // Update the newest existing backup instead of creating another file.
        uploaded = await api.files.update(
          metadata,
          canonicalId,
          uploadMedia: media,
          $fields: 'id,name,mimeType,size,createdTime,modifiedTime',
        );
      }

      if (uploaded.id == null) {
        throw StateError('Upload selesai tetapi ID file tidak ditemukan.');
      }

      // Only trash duplicates after the replacement upload/update succeeded.
      // A cleanup failure must not invalidate the newly uploaded backup.
      for (final duplicate in existing.skip(1)) {
        final duplicateId = duplicate.id;
        if (duplicateId == null || duplicateId.isEmpty) continue;
        try {
          await api.files.update(
            drive.File()..trashed = true,
            duplicateId,
            $fields: 'id,trashed',
          );
        } catch (_) {
          // Leave an undeleted duplicate in Drive; the newest remains canonical.
        }
      }

      return uploaded;
    });
  }

  Future<List<drive.File>> listEncryptedBackups() async {
    return _withDriveApi((api) async {

      // Restore UI exposes only the latest canonical backup.
      return results.isEmpty ? <drive.File>[] : <drive.File>[results.first];

  }

  Future<Uint8List> downloadEncryptedBackup(String fileId) async {
    if (fileId.trim().isEmpty) {
      throw ArgumentError('ID file backup kosong.');
    }

    return _withDriveApi((api) async {
      final response = await api.files.get(
        fileId,
        downloadOptions: drive.DownloadOptions.fullMedia,
      );

      if (response is! drive.Media) {
        throw StateError('Google Drive tidak mengembalikan isi file.');
      }

      final builder = BytesBuilder(copy: false);

      await for (final chunk in response.stream) {
        builder.add(chunk);

        if (builder.length > _maximumBackupBytes) {
          throw const FormatException(
            'File backup melebihi batas ukuran 100 MB.',
          );
        }
      }

      final bytes = builder.takeBytes();
      _validateEncryptedFile(bytes);

      return bytes;
    });
  }

import 'services/google_drive_service.dart';

void main() {
  WidgetsFlutterBinding.ensureInitialized();
  runApp(const BukuUtangApp());

  // Restore Google session without delaying offline-first app startup.
  unawaited(
    GoogleDriveService.instance.restoreAccount().then<void>(
      (_) {},
      onError: (Object error, StackTrace stack) {
        debugPrint('Pemulihan sesi Google saat startup gagal: $error');
      },
    ),
  );
}

class BukuUtangApp extends StatelessWidget {
  const BukuUtangApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Buku Utang Warung',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(
        useMaterial3: true,
        colorScheme: ColorScheme.fromSeed(seedColor: const Color(0xFF2E7D32)),
        visualDensity: VisualDensity.standard,
      ),
      home: const HomeScreen(),
    );
  }
}

===== BACKUP CRYPTO TEST =====
import 'dart:convert';

import 'package:flutter_test/flutter_test.dart';
import 'package:buku_utang_warung/services/backup_crypto_service.dart';

void main() {
  final crypto = BackupCryptoService.instance;

  final sampleBackup = <String, Object?>{
    'format': 'buku_utang_warung_backup',
    'version': 1,
    'exported_at': '2026-11-14T00:00:00.000Z',
    'customers': [
      {'id': 1, 'name': 'Pelanggan Uji'},
    ],
    'transactions': [
      {'id': 1, 'customer_id': 1, 'amount': 25000},
    ],
  };

  test('Backup dapat dienkripsi dan dibuka kembali', () async {
    const password = 'WarungAman2026';

    final encrypted = await crypto.encryptBackup(
      sampleBackup,
      password,
    );

    expect(encrypted, isNotEmpty);
    expect(utf8.decode(encrypted), isNot(contains('Pelanggan Uji')));

    final restored = await crypto.decryptBackup(
      encrypted,
      password,
    );

    expect(restored['customers'], sampleBackup['customers']);
    expect(restored['transactions'], sampleBackup['transactions']);
  });

  test('Password baru kurang dari 12 karakter ditolak', () async {
    expect(
      () => crypto.encryptBackup(sampleBackup, '12345678901'),
      throwsArgumentError,
    );
  });

  test('Password 8 karakter lolos validasi restore lama', () async {
    const legacyPassword = 'Lama2026';
    final encrypted = await crypto.encryptBackup(
      sampleBackup,
      'PasswordBaru2026',
    );

    // The decrypt path retains the legacy minimum-length rule.
    final restored = await crypto.decryptBackup(
      encrypted,
      'PasswordBaru2026',
    );
    expect(restored['customers'], sampleBackup['customers']);

    // An 8-character password remains accepted by the decrypt validator.
    // This checks the compatibility boundary without storing any secret.
    expect(
      () => crypto.decryptBackup(encrypted, legacyPassword),
      throwsFormatException,
    );
  });

  test('Password salah tidak dapat membuka backup', () async {
    final encrypted = await crypto.encryptBackup(
      sampleBackup,
      'PasswordBenar2026',
    );

    expect(
      () => crypto.decryptBackup(encrypted, 'PasswordSalah2026'),
      throwsFormatException,
    );
  });

  test('Isi ciphertext yang diubah akan ditolak', () async {
    final encrypted = await crypto.encryptBackup(
      sampleBackup,
      'PasswordBenar2026',
    );

    final envelope =
        jsonDecode(utf8.decode(encrypted)) as Map<String, dynamic>;

    final ciphertext = base64Decode(envelope['ciphertext'] as String);
    ciphertext[0] ^= 1;
    envelope['ciphertext'] = base64Encode(ciphertext);

    final tampered = utf8.encode(jsonEncode(envelope));

    expect(
      () => crypto.decryptBackup(tampered, 'PasswordBenar2026'),
      throwsFormatException,
    );
  });
}
