class CustomerScreen extends StatefulWidget {
  final int customerId;
  const CustomerScreen({super.key, required this.customerId});

  @override
  State<CustomerScreen> createState() => _CustomerScreenState();

class _CustomerScreenState extends State<CustomerScreen> {
  Customer? _customer;
  List<Txn> _txns = [];
  bool _loading = true;

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<void> _load() async {
    final db = AppDb.instance;
    final c = await db.customer(widget.customerId);
    final t = await db.txnsFor(widget.customerId);
    if (!mounted) return;
    if (c == null) {
      Navigator.pop(context);
      return;
    }
    setState(() {
      _customer = c;
      _txns = t;
      _loading = false;
    });
  }

  void _snack(String msg) {
    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(SnackBar(content: Text(msg)));
  }

  Future<void> _addTxn(TxnType type) async {
    final c = _customer!;
    final oldBalance = computeBalance(_txns);

    final txn = await showTxnForm(
      context,
      customer: c,
      type: type,
      balance: oldBalance,
    );
    if (txn == null) return;

    await AppDb.instance.insertTxn(txn);
    await _load();

    if (!mounted) return;

    final newBalance = computeBalance(_txns);
    _snack(type == TxnType.utang ? 'Utang dicatat' : 'Pembayaran dicatat');

    if (type == TxnType.bayar && oldBalance > 0 && newBalance <= 0) {
      await _sendSettlementReceipt();
    }
  }

  Future<void> _sendSettlementReceipt() async {
    final c = _customer!;
    final cycle = latestClosedDebtCycle(_txns);
    if (cycle.isEmpty) return;

    final send = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('Utang sudah lunas'),
        content: const Text(
          'Saldo pelanggan sudah lunas. Kirim bukti pelunasan '
          'beserta seluruh transaksi dalam siklus ini melalui WhatsApp?',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx, false),
            child: const Text('Nanti'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(ctx, true),
            child: const Text('Kirim WhatsApp'),
          ),
        ],
      ),
    );

    if (send != true || !mounted) return;

    final msg = buildSettlementReceipt(name: c.name, txns: cycle);
    var opened = false;

    try {
      opened = await launchUrl(
        waUri(c.phone, msg),
        mode: LaunchMode.externalApplication,
      );
    } catch (_) {
      opened = false;

===== SIKLUS UTANG =====
/// Mengurutkan transaksi berdasarkan waktu, lalu ID untuk waktu yang sama.
List<Txn> _orderedTxns(List<Txn> txns) {
  return [...txns]..sort((a, b) {
    final byDate = a.date.compareTo(b.date);
    return byDate != 0 ? byDate : (a.id ?? 0).compareTo(b.id ?? 0);
  });
}

/// Transaksi pada siklus utang yang masih aktif.
List<Txn> activeDebtCycle(List<Txn> txns) {
  final cycle = <Txn>[];
  var balance = 0;

  for (final t in _orderedTxns(txns)) {
    cycle.add(t);
    balance += t.type == TxnType.utang ? t.amount : -t.amount;

    if (balance <= 0) {
      cycle.clear();
      balance = 0;
    }
  }

  return cycle;
}

/// Transaksi pada siklus terakhir yang sudah lunas.
List<Txn> latestClosedDebtCycle(List<Txn> txns) {
  final cycle = <Txn>[];
  List<Txn> latestClosed = [];
  var balance = 0;

  for (final t in _orderedTxns(txns)) {
    cycle.add(t);
    balance += t.type == TxnType.utang ? t.amount : -t.amount;

    if (balance <= 0) {
      if (cycle.any((item) => item.type == TxnType.utang)) {
        latestClosed = List<Txn>.from(cycle);
      }
      cycle.clear();
      balance = 0;
    }
  }

  return latestClosed;
}
