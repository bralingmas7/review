===== BACKUP STATUS SERVICE =====
     1	import 'dart:async';
     2	
     3	import 'package:flutter/foundation.dart';
     4	import 'package:shared_preferences/shared_preferences.dart';
     5	
     6	/// Melacak apakah ada perubahan data yang belum di-backup, terpisah untuk
     7	/// backup lokal (file di HP) dan Google Drive.
     8	///
     9	/// Nilai notifier: null = belum diketahui, true = ada perubahan belum
    10	/// di-backup, false = sudah di-backup.
    11	class BackupStatus {
    12	  BackupStatus._();
    13	
    14	  static final BackupStatus instance = BackupStatus._();
    15	
    16	  // Kunci lama dipertahankan untuk Google Drive agar status lama tidak hilang.
    17	  static const String _driveKey = 'backup_dirty';
    18	  static const String _localKey = 'backup_local_dirty';
    19	
    20	  final ValueNotifier<bool?> driveDirty = ValueNotifier<bool?>(null);
    21	  final ValueNotifier<bool?> localDirty = ValueNotifier<bool?>(null);
    22	
    23	  bool _loaded = false;
    24	  int _revision = 0;
    25	
    26	  /// Naik setiap data berubah. Dipakai agar perubahan yang terjadi saat
    27	  /// backup berlangsung tidak ikut dianggap sudah ter-backup.
    28	  int get revision => _revision;
    29	
    30	  /// Dipanggil sekali saat app dibuka. Panggilan berikutnya diabaikan.
    31	  Future<void> load({required bool hasData}) async {
    32	    if (_loaded) return;
    33	    _loaded = true;
    34	
    35	    try {
    36	      final prefs = SharedPreferencesAsync();
    37	      final drive = await prefs.getBool(_driveKey);
    38	      final local = await prefs.getBool(_localKey);
    39	
    40	      // Pertama kali: kalau sudah ada data tapi belum pernah di-backup,
    41	      // anggap belum ter-backup. Kalau masih kosong, anggap aman.
    42	      driveDirty.value ??= drive ?? hasData;
    43	      localDirty.value ??= local ?? hasData;
    44	
    45	      if (drive == null) await _persist(_driveKey, driveDirty.value ?? hasData);
    46	      if (local == null) await _persist(_localKey, localDirty.value ?? hasData);
    47	    } catch (e) {
    48	      debugPrint('Gagal memuat status backup: $e');
    49	      driveDirty.value ??= hasData;
    50	      localDirty.value ??= hasData;
    51	    }
    52	  }
    53	
    54	  void markDirty() {
    55	    _revision++;
    56	    driveDirty.value = true;
    57	    localDirty.value = true;
    58	    unawaited(_persist(_driveKey, true));
    59	    unawaited(_persist(_localKey, true));
    60	  }
    61	
    62	  /// Tandai Google Drive sudah ter-backup, hanya jika data tidak berubah
    63	  /// sejak [revision] dicatat (sebelum backup dibuat).
    64	  Future<void> markDriveBackedUp(int revision) async {
    65	    if (revision != _revision) return;
    66	    driveDirty.value = false;
    67	    await _persist(_driveKey, false);
    68	  }
    69	
    70	  /// Sama seperti [markDriveBackedUp], untuk backup file lokal.
    71	  Future<void> markLocalBackedUp(int revision) async {
    72	    if (revision != _revision) return;
    73	    localDirty.value = false;
    74	    await _persist(_localKey, false);
    75	  }
    76	
    77	  /// Nama lama, tetap dipakai oleh alur backup Google Drive.
    78	  Future<void> markBackedUp(int revision) => markDriveBackedUp(revision);
    79	
    80	  Future<void> _persist(String key, bool value) async {
    81	    try {
    82	      await SharedPreferencesAsync().setBool(key, value);
    83	    } catch (e) {
    84	      debugPrint('Gagal menyimpan status backup: $e');
    85	    }
    86	  }
    87	}
