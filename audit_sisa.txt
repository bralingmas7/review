  final GlobalKey _imageKey = GlobalKey();
  Uint8List? _pngBytes;
  bool _working = false;

  @override

    super.initState();
    WidgetsBinding.instance.addPostFrameCallback((_) => _capture());
  }

  String get _fileName {
    final safeName = widget.customer.name
        .trim()
        .replaceAll(RegExp(r'[^a-zA-Z0-9_-]+'), '_');
    final kind = widget.settlement ? 'pelunasan' : 'rekap';
    final now = DateTime.now();
    final stamp = '${now.year}${now.month.toString().padLeft(2, '0')}'
        '${now.day.toString().padLeft(2, '0')}_'
        '${now.hour.toString().padLeft(2, '0')}'
        '${now.minute.toString().padLeft(2, '0')}'
        '${now.second.toString().padLeft(2, '0')}';
    return '${kind}_${safeName}_$stamp.png';
  }

  Future<void> _capture() async {
    try {
      await WidgetsBinding.instance.endOfFrame;
      final boundary = _imageKey.currentContext?.findRenderObject()
          as RenderRepaintBoundary?;
      if (boundary == null || !boundary.hasSize) {
        throw Exception('Pratinjau gambar belum siap.');
      }

      final ui.Image image = await boundary.toImage(pixelRatio: 3);
      try {
        final data = await image.toByteData(format: ui.ImageByteFormat.png);
        if (data == null) throw Exception('Gagal membuat data PNG.');
        if (!mounted) return;
        setState(() => _pngBytes = data.buffer.asUint8List());
      } finally {
        image.dispose();
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Gagal menyiapkan PNG: $e')),
        );
      }
    }
  }

  Future<void> _save() async {
    final bytes = _pngBytes;
    if (bytes == null || _working) return;

    setState(() => _working = true);
    try {
      await Gal.putImageBytes(
        bytes,
        name: _fileName.replaceAll('.png', ''),
      );
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('PNG berhasil disimpan ke galeri.')),
        );
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Gagal menyimpan PNG: $e')),
        );
      }
    } finally {
      if (mounted) setState(() => _working = false);
    }
  }

  Future<void> _share() async {
    final bytes = _pngBytes;
    if (bytes == null || _working) return;

    setState(() => _working = true);
    try {
      await Share.shareXFiles(
        [
          XFile.fromData(
            bytes,
            mimeType: 'image/png',
            name: _fileName,
          ),
        ],
        fileNameOverrides: [_fileName],
      );
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Gagal membagikan PNG: $e')),
        );
      }
    } finally {
      if (mounted) setState(() => _working = false);
    }
  }

===== SHOP SETTINGS SERVICE =====
import 'package:shared_preferences/shared_preferences.dart';

class ShopMessageSettings {
  const ShopMessageSettings({
    this.shopName = '',
    this.header = '',
    this.footer = '',
  });

  final String shopName;
  final String header;
  final String footer;

  static const _nameKey = 'shop_message_name';
  static const _headerKey = 'shop_message_header';
  static const _footerKey = 'shop_message_footer';

  static Future<ShopMessageSettings> load() async {
    final prefs = SharedPreferencesAsync();
    return ShopMessageSettings(
      shopName: await prefs.getString(_nameKey) ?? '',
      header: await prefs.getString(_headerKey) ?? '',
      footer: await prefs.getString(_footerKey) ?? '',
    );
  }

  Future<void> save() async {
    final prefs = SharedPreferencesAsync();
    await prefs.setString(_nameKey, shopName.trim());
    await prefs.setString(_headerKey, header.trim());
    await prefs.setString(_footerKey, footer.trim());
  }
}
