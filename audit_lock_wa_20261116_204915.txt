import 'dart:convert';
import 'dart:math';
import 'dart:typed_data';

import 'package:cryptography/cryptography.dart';
import 'package:flutter/foundation.dart';
import 'package:local_auth/local_auth.dart';
import 'package:shared_preferences/shared_preferences.dart';

enum PinResult { ok, wrong, locked }

class PinCheck {
  const PinCheck(this.result, {this.remainingAttempts = 0, this.wait});

  final PinResult result;
  final int remainingAttempts;
  final Duration? wait;
}

/// Kunci aplikasi: PIN 4-6 angka (disimpan sebagai hash PBKDF2 + salt acak)
/// dan opsi sidik jari / kunci layar lewat local_auth.
class AppLockService {
  AppLockService._();

  static final AppLockService instance = AppLockService._();

  static const int minPinLength = 4;
  static const int maxPinLength = 6;
  static const int maxFreeAttempts = 5;

  static const int _iterations = 50000;

  static const String _enabledKey = 'app_lock_enabled';
  static const String _saltKey = 'app_lock_salt';
  static const String _hashKey = 'app_lock_hash';
  static const String _bioKey = 'app_lock_bio';
  static const String _failsKey = 'app_lock_fails';
  static const String _untilKey = 'app_lock_until';

  final Pbkdf2 _kdf = Pbkdf2(
    macAlgorithm: Hmac.sha256(),
    iterations: _iterations,
    bits: 256,
  );

  final LocalAuthentication _auth = LocalAuthentication();

  bool _loaded = false;
  bool _enabled = false;
  bool _biometric = false;

  bool get enabled => _enabled;
  bool get biometricEnabled => _enabled && _biometric;

  Future<void> load() async {
    if (_loaded) return;

    try {
      final prefs = SharedPreferencesAsync();
      _enabled = (await prefs.getBool(_enabledKey)) ?? false;
      _biometric = (await prefs.getBool(_bioKey)) ?? false;

      // Tanpa hash tersimpan, kunci tidak mungkin dibuka. Anggap mati.
      if (_enabled && (await prefs.getString(_hashKey)) == null) {
        _enabled = false;
      }
    } catch (e) {
      debugPrint('Gagal memuat kunci aplikasi: $e');
      _enabled = false;
      _biometric = false;
    }

    _loaded = true;
  }

  Future<Uint8List> _hash(String pin, List<int> salt) async {
    final key = await _kdf.deriveKeyFromPassword(password: pin, nonce: salt);
    return Uint8List.fromList(await key.extractBytes());
  }

  bool _sameBytes(List<int> a, List<int> b) {
    if (a.length != b.length) return false;
    var diff = 0;
    for (var i = 0; i < a.length; i++) {
      diff |= a[i] ^ b[i];
    }
    return diff == 0;
  }

  Future<void> setPin(String pin) async {
    if (!RegExp(r'^\d{4,6}$').hasMatch(pin)) {
      throw ArgumentError('PIN harus 4 sampai 6 angka.');
    }

    final random = Random.secure();
    final salt = Uint8List.fromList(
      List<int>.generate(16, (_) => random.nextInt(256)),
    );
    final hash = await _hash(pin, salt);

    final prefs = SharedPreferencesAsync();
    await prefs.setString(_saltKey, base64Encode(salt));
    await prefs.setString(_hashKey, base64Encode(hash));
    await prefs.setBool(_enabledKey, true);
    await prefs.setInt(_failsKey, 0);
    await prefs.setInt(_untilKey, 0);

    _enabled = true;
    _loaded = true;
  }

  Future<Duration?> lockoutRemaining() async {
    try {
      final until = (await SharedPreferencesAsync().getInt(_untilKey)) ?? 0;
      final now = DateTime.now().millisecondsSinceEpoch;
      if (until <= now) return null;
      return Duration(seconds: ((until - now) / 1000).ceil());
    } catch (_) {
      return null;
    }
  }

  Future<PinCheck> verifyPin(String pin) async {
    final prefs = SharedPreferencesAsync();

    final remaining = await lockoutRemaining();
    if (remaining != null) return PinCheck(PinResult.locked, wait: remaining);

    final saltText = await prefs.getString(_saltKey);
    final hashText = await prefs.getString(_hashKey);
    if (saltText == null || hashText == null) {
      return const PinCheck(PinResult.wrong);
    }

    final actual = await _hash(pin, base64Decode(saltText));

    if (_sameBytes(actual, base64Decode(hashText))) {
      await prefs.setInt(_failsKey, 0);
      await prefs.setInt(_untilKey, 0);
      return const PinCheck(PinResult.ok);
    }

    final fails = ((await prefs.getInt(_failsKey)) ?? 0) + 1;
    await prefs.setInt(_failsKey, fails);

    if (fails % maxFreeAttempts == 0) {
      // Jeda makin lama: 30 dtk, 60, 120, 240, 480, lalu maksimum 600 dtk.
      final round = fails ~/ maxFreeAttempts;
      final seconds = min(30 * pow(2, round - 1).toInt(), 600);
      final until = DateTime.now().millisecondsSinceEpoch + seconds * 1000;
      await prefs.setInt(_untilKey, until);
      return PinCheck(PinResult.locked, wait: Duration(seconds: seconds));
    }

    return PinCheck(
      PinResult.wrong,
      remainingAttempts: maxFreeAttempts - (fails % maxFreeAttempts),
    );
  }

  Future<void> disable() async {
    try {
      final prefs = SharedPreferencesAsync();
      await prefs.setBool(_enabledKey, false);
      await prefs.setBool(_bioKey, false);
      await prefs.remove(_saltKey);
      await prefs.remove(_hashKey);
      await prefs.setInt(_failsKey, 0);
      await prefs.setInt(_untilKey, 0);
    } catch (e) {
      debugPrint('Gagal mematikan kunci aplikasi: $e');
    }

    _enabled = false;
    _biometric = false;
  }

  Future<void> setBiometric(bool value) async {
    await SharedPreferencesAsync().setBool(_bioKey, value);
    _biometric = value;
  }

  /// Ada sidik jari / wajah yang terdaftar di HP.
  Future<bool> biometricAvailable() async {
    try {
      final types = await _auth.getAvailableBiometrics();
      return types.isNotEmpty;
    } catch (_) {
      return false;
    }
  }

  /// [biometricOnly] false mengizinkan PIN / pola / kata sandi layar HP.
  Future<bool> authenticate({
    required String reason,
    required bool biometricOnly,
  }) async {
    try {
      return await _auth.authenticate(
        localizedReason: reason,
        biometricOnly: biometricOnly,
        persistAcrossBackgrounding: true,
      );
    } catch (e) {
      debugPrint('Autentikasi perangkat gagal: $e');
      return false;
    }
  }
}

import 'package:flutter/material.dart';

import '../screens/lock_screen.dart';

/// Membungkus seluruh app. Menampilkan layar kunci saat app dibuka, dan saat
/// kembali dari latar belakang setelah lebih dari [_grace].
class AppLockGate extends StatefulWidget {
  const AppLockGate({super.key, required this.child});

  final Widget child;

  @override
  State<AppLockGate> createState() => _AppLockGateState();
}

class _AppLockGateState extends State<AppLockGate> with WidgetsBindingObserver {
  // Pemilih akun Google dan pemilih file bisa membuat app ke latar belakang
  // sebentar. Jeda ini mencegah app terkunci di tengah proses itu.
  static const Duration _grace = Duration(seconds: 60);

  bool _ready = false;
  bool _locked = false;
  DateTime? _leftAt;

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _init();
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  Future<void> _init() async {
    await AppLockService.instance.load();
    if (!mounted) return;
    setState(() {
      _locked = AppLockService.instance.enabled;
      _ready = true;
    });
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.paused ||
        state == AppLifecycleState.hidden) {
      _leftAt ??= DateTime.now();
      return;
    }

    if (state == AppLifecycleState.resumed) {
      final left = _leftAt;
      _leftAt = null;

      if (left == null || !_ready || _locked) return;
      if (!AppLockService.instance.enabled) return;

      if (DateTime.now().difference(left) >= _grace) {
        setState(() => _locked = true);
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final hidden = !_ready || _locked;

    return Stack(
      fit: StackFit.expand,
      children: [
        ExcludeSemantics(excluding: hidden, child: widget.child),
        if (!_ready)
          Positioned.fill(
            child: ColoredBox(
              color: Theme.of(context).scaffoldBackgroundColor,
            ),
          )
        else if (_locked)
          Positioned.fill(
            child: LockScreen(
              key: const ValueKey('app-lock-screen'),
              onUnlocked: () => setState(() => _locked = false),
            ),
          ),
      ],
    );
  }
}

    final remaining = await _lock.lockoutRemaining();
    if (remaining != null && mounted) _startWait(remaining);
  }

  void _startWait(Duration duration) {
    _ticker?.cancel();

    setState(() {
      _wait = duration;
      _busy = false;
      _pin = '';
      _message = null;
    });

    _ticker = Timer.periodic(const Duration(seconds: 1), (timer) {
      if (!mounted) {
        timer.cancel();
        return;
      }

      final current = _wait;
      if (current == null) {
        timer.cancel();
        return;
      }

      final next = current - const Duration(seconds: 1);
      if (next <= Duration.zero) {
        timer.cancel();
        setState(() => _wait = null);
      } else {
        setState(() => _wait = next);
      }
    });
  }

  Future<void> _useBiometric() async {
    if (!mounted || _busy || _wait != null) return;

    setState(() => _busy = true);

    final ok = await _lock.authenticate(
      reason: 'Buka Buku Utang Warung',
      biometricOnly: true,
    );

    if (!mounted) return;

    if (ok) {
      widget.onUnlocked();
    } else {
      setState(() => _busy = false);
    }
  }

  Future<void> _submit() async {
    if (_busy || _wait != null) return;
    if (_pin.length < AppLockService.minPinLength) return;

    setState(() => _busy = true);

    final check = await _lock.verifyPin(_pin);

    if (!mounted) return;

    switch (check.result) {
      case PinResult.ok:
        widget.onUnlocked();
        break;
      case PinResult.locked:
        _startWait(check.wait ?? const Duration(seconds: 30));
        break;
      case PinResult.wrong:
        setState(() {
          _busy = false;
          _pin = '';
          _message = 'PIN salah. Sisa percobaan: ${check.remainingAttempts}';
        });
        break;
    }
  }

  Future<void> _forgotPin() async {
    if (_busy) return;

    setState(() {
      _busy = true;
      _message = null;
    });

    final ok = await _lock.authenticate(
      reason: 'Verifikasi dengan kunci layar HP untuk mereset PIN',
      biometricOnly: false,
    );

    if (!mounted) return;

    setState(() {
      _busy = false;
      if (ok) {
        _resetPending = true;
      } else {
        _message = 'Verifikasi kunci layar gagal atau dibatalkan. '
            'HP harus punya kunci layar (PIN, pola, atau kata sandi).';
      }
    });
  }

  Future<void> _disableAndContinue() async {
    await _lock.disable();
    widget.onUnlocked();
  }

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);

    final Widget content;

    if (_resetPending) {
      content = Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          const Text(
            'Kunci layar HP terverifikasi.\n\n'
            'Kunci aplikasi akan dimatikan. Kamu bisa membuat PIN baru di '
            'Pengaturan. Data tidak dihapus.',
            textAlign: TextAlign.center,
          ),
          const SizedBox(height: 24),
          FilledButton(
            onPressed: _disableAndContinue,
            child: const Text('Matikan kunci dan lanjut'),
          ),
          TextButton(
            onPressed: () => setState(() => _resetPending = false),
            child: const Text('Batal'),
          ),
        ],
      );
    } else {
      final wait = _wait;

      content = Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          Text(
            wait != null
                ? 'Terlalu banyak percobaan salah.\n'
                    'Coba lagi dalam ${wait.inSeconds} detik.'
                : 'Masukkan PIN',
            textAlign: TextAlign.center,
            style: theme.textTheme.titleMedium,
          ),
          SizedBox(
            height: 40,
            child: Center(
              child: _message == null
                  ? null
                  : Text(
                      _message!,
                      textAlign: TextAlign.center,
                      style: TextStyle(color: theme.colorScheme.error),
                    ),
            ),
          ),
          PinPad(
            pin: _pin,
            enabled: !_busy && wait == null,
            onChanged: (value) => setState(() {
              _pin = value;
              _message = null;
            }),

enum PinMode { create, verify }

/// Layar PIN untuk Pengaturan.
/// create: minta PIN dua kali, hasilnya (String) dikembalikan lewat pop.
/// verify: cek PIN saat ini, hasilnya true dikembalikan lewat pop.
class PinScreen extends StatefulWidget {
  const PinScreen({super.key, required this.mode, required this.title});

  final PinMode mode;
  final String title;

  @override
  State<PinScreen> createState() => _PinScreenState();
}

class _PinScreenState extends State<PinScreen> {
  String _pin = '';
  String? _first;
  String? _message;
  bool _busy = false;

  String get _prompt {
    if (widget.mode == PinMode.verify) return 'Masukkan PIN saat ini';
    return _first == null ? 'Buat PIN (4-6 angka)' : 'Ulangi PIN';
  }

  Future<void> _submit() async {
    if (_busy || _pin.length < AppLockService.minPinLength) return;

    if (widget.mode == PinMode.create) {
      if (_first == null) {
        setState(() {
          _first = _pin;
          _pin = '';
          _message = null;
        });
        return;
      }

      if (_first == _pin) {
        Navigator.pop(context, _pin);
      } else {
        setState(() {
          _first = null;
          _pin = '';
          _message = 'PIN tidak sama. Ulangi dari awal.';
        });
      }
      return;
    }

    setState(() => _busy = true);

    final check = await AppLockService.instance.verifyPin(_pin);

    if (!mounted) return;

    switch (check.result) {
      case PinResult.ok:
        Navigator.pop(context, true);
        break;
      case PinResult.locked:
        setState(() {
          _busy = false;
          _pin = '';
          _message = 'Terlalu banyak percobaan. '
              'Coba lagi dalam ${check.wait?.inSeconds ?? 30} detik.';
        });
        break;
      case PinResult.wrong:
        setState(() {
          _busy = false;
          _pin = '';
          _message = 'PIN salah. Sisa percobaan: ${check.remainingAttempts}';
        });
        break;
    }
  }

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);

    return Scaffold(
      appBar: AppBar(title: Text(widget.title)),

import 'package:flutter/material.dart';

import '../services/app_lock_service.dart';

/// Tombol angka untuk memasukkan PIN. Tidak memakai keyboard bawaan.
class PinPad extends StatelessWidget {
  const PinPad({
    super.key,
    required this.pin,
    required this.onChanged,
    required this.onSubmit,
    this.onBiometric,
    this.enabled = true,
  });

  final String pin;
  final ValueChanged<String> onChanged;
  final VoidCallback onSubmit;
  final VoidCallback? onBiometric;
  final bool enabled;

  void _add(String digit) {
    if (!enabled || pin.length >= AppLockService.maxPinLength) return;
    onChanged(pin + digit);
  }

  void _back() {
    if (!enabled || pin.isEmpty) return;
    onChanged(pin.substring(0, pin.length - 1));
  }

  Widget _key(Widget child, VoidCallback? onTap) {
    return SizedBox(
      width: 76,
      height: 68,
      child: InkWell(
        customBorder: const CircleBorder(),
        onTap: enabled ? onTap : null,
        child: Center(child: child),
      ),
    );
  }

  Widget _digit(String d) => _key(
        Text(d, style: const TextStyle(fontSize: 26)),
        () => _add(d),
      );

  @override
  Widget build(BuildContext context) {
    final primary = Theme.of(context).colorScheme.primary;

    return Column(
      mainAxisSize: MainAxisSize.min,
      children: [
        Row(
          mainAxisAlignment: MainAxisAlignment.center,
          children: List.generate(AppLockService.maxPinLength, (i) {
            return Container(
              margin: const EdgeInsets.symmetric(horizontal: 6),
              width: 14,
              height: 14,
              decoration: BoxDecoration(
                shape: BoxShape.circle,
                color: i < pin.length ? primary : Colors.transparent,
                border: Border.all(color: primary, width: 1.5),
              ),
            );
          }),
        ),
        const SizedBox(height: 20),
        for (final row in const [
          ['1', '2', '3'],
          ['4', '5', '6'],
          ['7', '8', '9'],
        ])
          Row(
            mainAxisSize: MainAxisSize.min,
            children: [for (final d in row) _digit(d)],
          ),
        Row(
          mainAxisSize: MainAxisSize.min,
          children: [
            _key(
              onBiometric == null
                  ? const SizedBox.shrink()
                  : const Icon(Icons.fingerprint, size: 30),
              onBiometric,
            ),
            _digit('0'),
            _key(const Icon(Icons.backspace_outlined), _back),
          ],
        ),
        const SizedBox(height: 12),
        SizedBox(
          width: 228,
          child: FilledButton(
            onPressed: enabled && pin.length >= AppLockService.minPinLength
                ? onSubmit
                : null,
            child: const Text('OK'),
          ),
        ),
      ],
    );
  }
}

import 'package:flutter/material.dart';

/// Kartu pengaturan kunci aplikasi untuk layar Pengaturan.
class AppLockSettingsCard extends StatefulWidget {
  const AppLockSettingsCard({super.key});

  @override
  State<AppLockSettingsCard> createState() => _AppLockSettingsCardState();
}

class _AppLockSettingsCardState extends State<AppLockSettingsCard> {
  final AppLockService _lock = AppLockService.instance;

  bool _enabled = false;
  bool _biometric = false;
  bool _biometricAvailable = false;
  bool _busy = false;

  @override
  void initState() {
    super.initState();
    _refresh();
  }

  Future<void> _refresh() async {
    await _lock.load();
    final available = await _lock.biometricAvailable();
    if (!mounted) return;

    setState(() {
      _enabled = _lock.enabled;
      _biometric = _lock.biometricEnabled;
      _biometricAvailable = available;
    });
  }

  void _snack(String text) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(text)));
  }

  Future<String?> _createPin(String title) {
    return Navigator.push<String>(
      context,
      MaterialPageRoute(
        builder: (_) => PinScreen(mode: PinMode.create, title: title),
      ),
    );
  }

  Future<bool> _verifyPin(String title) async {
    final ok = await Navigator.push<bool>(
      context,
      MaterialPageRoute(
        builder: (_) => PinScreen(mode: PinMode.verify, title: title),
      ),
    );
    return ok == true;
  }

  Future<void> _toggleLock(bool on) async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      if (on) {
        final pin = await _createPin('Buat PIN');
        if (pin == null) return;

        await _lock.setPin(pin);
        _snack('Kunci aplikasi aktif.');
      } else {
        final ok = await _verifyPin('Masukkan PIN untuk mematikan kunci');
        if (!ok) return;

        await _lock.disable();
        _snack('Kunci aplikasi dimatikan.');
      }
    } catch (e) {
      _snack('Gagal mengubah kunci aplikasi: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
      await _refresh();
    }
  }

  Future<void> _changePin() async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      final ok = await _verifyPin('Masukkan PIN saat ini');
      if (!ok || !mounted) return;

      final pin = await _createPin('Buat PIN baru');
      if (pin == null) return;

      await _lock.setPin(pin);
      _snack('PIN berhasil diubah.');
    } catch (e) {
      _snack('Gagal mengubah PIN: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
      await _refresh();
    }
  }

  Future<void> _toggleBiometric(bool on) async {
    if (_busy) return;
    setState(() => _busy = true);

    try {
      if (on) {
        final ok = await _lock.authenticate(
          reason: 'Aktifkan buka dengan sidik jari',
          biometricOnly: true,
        );

        if (!ok) {
          _snack('Verifikasi sidik jari gagal atau dibatalkan.');
          return;
        }
      }

      await _lock.setBiometric(on);
    } catch (e) {
      _snack('Gagal mengubah pengaturan sidik jari: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
      await _refresh();
    }
  }

  @override
  Widget build(BuildContext context) {
    return Card(
      child: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            const ListTile(
              contentPadding: EdgeInsets.zero,
              leading: Icon(Icons.lock_outline),
              title: Text('Kunci Aplikasi'),
              subtitle: Text('Minta PIN atau sidik jari saat app dibuka.'),
            ),
            SwitchListTile(
              contentPadding: EdgeInsets.zero,
              title: const Text('Kunci dengan PIN'),
              value: _enabled,
              onChanged: _busy ? null : _toggleLock,
            ),
            if (_enabled) ...[
              SwitchListTile(
                contentPadding: EdgeInsets.zero,
                title: const Text('Buka dengan sidik jari'),
                subtitle: Text(
                  _biometricAvailable
                      ? 'PIN tetap bisa dipakai kapan saja.'
                      : 'Belum ada sidik jari yang terdaftar di HP.',
                ),
                value: _biometric,
                onChanged: _busy || !_biometricAvailable ? null : _toggleBiometric,
              ),
              ListTile(
                contentPadding: EdgeInsets.zero,
                title: const Text('Ubah PIN'),
                trailing: const Icon(Icons.chevron_right),
                onTap: _busy ? null : _changePin,
              ),
            ],
            const SizedBox(height: 8),
            const Text(
              'Kalau lupa PIN, ketuk "Lupa PIN?" di layar kunci lalu '
              'verifikasi dengan kunci layar HP. Data tidak dihapus. '
              'Kunci ini menahan orang lain membuka app, tetapi tidak '
              'mengenkripsi database di HP.',
              style: TextStyle(fontSize: 12),
            ),
          ],
        ),
      ),
    );
  }
}
