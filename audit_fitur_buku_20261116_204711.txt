import 'dart:async';

import 'package:flutter/foundation.dart';
import 'package:shared_preferences/shared_preferences.dart';

/// Melacak apakah ada perubahan data yang belum di-backup, terpisah untuk
/// backup lokal (file di HP) dan Google Drive.
///
/// Nilai notifier: null = belum diketahui, true = ada perubahan belum
/// di-backup, false = sudah di-backup.
class BackupStatus {
  BackupStatus._();

  static final BackupStatus instance = BackupStatus._();

  // Kunci lama dipertahankan untuk Google Drive agar status lama tidak hilang.
  static const String _driveKey = 'backup_dirty';
  static const String _localKey = 'backup_local_dirty';

  final ValueNotifier<bool?> driveDirty = ValueNotifier<bool?>(null);
  final ValueNotifier<bool?> localDirty = ValueNotifier<bool?>(null);

  bool _loaded = false;
  int _revision = 0;

  /// Naik setiap data berubah. Dipakai agar perubahan yang terjadi saat
  /// backup berlangsung tidak ikut dianggap sudah ter-backup.
  int get revision => _revision;

  /// Dipanggil sekali saat app dibuka. Panggilan berikutnya diabaikan.
  Future<void> load({required bool hasData}) async {
    if (_loaded) return;
    _loaded = true;

    try {
      final prefs = SharedPreferencesAsync();
      final drive = await prefs.getBool(_driveKey);
      final local = await prefs.getBool(_localKey);

      // Pertama kali: kalau sudah ada data tapi belum pernah di-backup,
      // anggap belum ter-backup. Kalau masih kosong, anggap aman.
      driveDirty.value ??= drive ?? hasData;
      localDirty.value ??= local ?? hasData;

      if (drive == null) await _persist(_driveKey, driveDirty.value ?? hasData);
      if (local == null) await _persist(_localKey, localDirty.value ?? hasData);
    } catch (e) {
      debugPrint('Gagal memuat status backup: $e');
      driveDirty.value ??= hasData;
      localDirty.value ??= hasData;
    }
  }

  void markDirty() {
    _revision++;
    driveDirty.value = true;
    localDirty.value = true;
    unawaited(_persist(_driveKey, true));
    unawaited(_persist(_localKey, true));
  }

  /// Tandai Google Drive sudah ter-backup, hanya jika data tidak berubah
  /// sejak [revision] dicatat (sebelum backup dibuat).
  Future<void> markDriveBackedUp(int revision) async {
    if (revision != _revision) return;
    driveDirty.value = false;
    await _persist(_driveKey, false);
  }

  /// Sama seperti [markDriveBackedUp], untuk backup file lokal.
  Future<void> markLocalBackedUp(int revision) async {
    if (revision != _revision) return;
    localDirty.value = false;
    await _persist(_localKey, false);
  }

  /// Nama lama, tetap dipakai oleh alur backup Google Drive.
  Future<void> markBackedUp(int revision) => markDriveBackedUp(revision);

  Future<void> _persist(String key, bool value) async {
    try {
      await SharedPreferencesAsync().setBool(key, value);
    } catch (e) {
      debugPrint('Gagal menyimpan status backup: $e');
    }
  }
}

import 'dart:async';

import 'package:flutter/material.dart';

import '../services/backup_status.dart';

/// Dua bulatan status di AppBar: ikon HP = backup lokal, ikon awan = Google
/// Drive. Merah/biru bergantian = belum di-backup. Hijau = sudah di-backup.
class BackupIndicator extends StatefulWidget {
  const BackupIndicator({super.key, this.onTap});

  final VoidCallback? onTap;

  @override
  State<BackupIndicator> createState() => _BackupIndicatorState();
}

class _BackupIndicatorState extends State<BackupIndicator> {
  Timer? _timer;
  bool _alt = false;

  @override
  void initState() {
    super.initState();
    BackupStatus.instance.localDirty.addListener(_onChanged);
    BackupStatus.instance.driveDirty.addListener(_onChanged);
    _syncTimer();
  }

  @override
  void dispose() {
    BackupStatus.instance.localDirty.removeListener(_onChanged);
    BackupStatus.instance.driveDirty.removeListener(_onChanged);
    _timer?.cancel();
    super.dispose();
  }

  bool get _anyDirty =>
      BackupStatus.instance.localDirty.value == true ||
      BackupStatus.instance.driveDirty.value == true;

  void _onChanged() {
    _syncTimer();
    if (mounted) setState(() {});
  }

  void _syncTimer() {
    if (_anyDirty && _timer == null) {
      _timer = Timer.periodic(const Duration(milliseconds: 600), (_) {
        if (mounted) setState(() => _alt = !_alt);
      });
    } else if (!_anyDirty && _timer != null) {
      _timer!.cancel();
      _timer = null;
      _alt = false;
    }
  }

  String _describe(bool? state) {
    if (state == null) return 'belum diketahui';
    return state ? 'ada perubahan belum di-backup' : 'sudah di-backup';
  }

  Color _colorFor(bool? state) {
    if (state == null) return Colors.grey;
    if (state) return _alt ? Colors.blue : Colors.red;
    return Colors.green;
  }

  Widget _item(IconData icon, bool? state) {
    final color = _colorFor(state);
    return Row(
      mainAxisSize: MainAxisSize.min,
      children: [
        Icon(icon, size: 18),
        const SizedBox(width: 4),
        AnimatedContainer(
          duration: const Duration(milliseconds: 250),
          width: 12,
          height: 12,
          decoration: BoxDecoration(
            color: color,
            shape: BoxShape.circle,
            boxShadow: [
              BoxShadow(
                color: color.withOpacity(0.5),
                blurRadius: 5,
                spreadRadius: 1,
              ),
            ],
          ),
        ),
      ],
    );
  }

  @override
  Widget build(BuildContext context) {
    final local = BackupStatus.instance.localDirty.value;
    final drive = BackupStatus.instance.driveDirty.value;

    // Status belum diketahui: jangan tampilkan apa pun supaya tidak salah.
    if (local == null && drive == null) return const SizedBox.shrink();

    return Tooltip(
      message: 'Backup lokal: ${_describe(local)}\n'
          'Backup Google Drive: ${_describe(drive)}',
      child: InkWell(
        onTap: widget.onTap,
        borderRadius: BorderRadius.circular(20),
        child: Padding(
          padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 8),
          child: Row(
            mainAxisSize: MainAxisSize.min,
            children: [
              _item(Icons.smartphone, local),
              const SizedBox(width: 10),
              _item(Icons.cloud_outlined, drive),
            ],
          ),
        ),
      ),
    );
  }
}
