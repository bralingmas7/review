for p in targets:
    dst = backup / p.relative_to(root)
    dst.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(p, dst)

for p in targets:
    dst = backup / p.relative_to(root)

    for p, content in data.items():
        tmp = p.with_name(p.name + ".patch_tmp")
        tmp.write_text(content)

    for p, tmp in staged:
        os.replace(tmp, p)
except Exception:
    for p, tmp in staged:
