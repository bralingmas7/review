import 'package:path/path.dart' as p;
import 'package:sqflite/sqflite.dart';

import 'models.dart';

class AppDb {
  AppDb._();
  static final AppDb instance = AppDb._();

  static const backupFormat = 'buku_utang_warung_backup';
  static const backupVersion = 1;

  Database? _db;

  Future<Database> get _database async => _db ??= await _open();

  Future<Database> _open() async {
    final path = p.join(await getDatabasesPath(), 'buku_utang.db');
    return openDatabase(
      path,
      version: 1,
      onConfigure: (db) async {
        await db.execute('PRAGMA foreign_keys = ON');
      },
      onCreate: (db, version) async {
        await db.execute('''
          CREATE TABLE customers(
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            phone TEXT NOT NULL DEFAULT '',
            note TEXT NOT NULL DEFAULT '',
            credit_limit INTEGER NOT NULL DEFAULT 0,
            created_at INTEGER NOT NULL
          )
        ''');
        await db.execute('''
          CREATE TABLE transactions(
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            customer_id INTEGER NOT NULL,
            type TEXT NOT NULL,
            amount INTEGER NOT NULL,
            note TEXT NOT NULL DEFAULT '',
            date INTEGER NOT NULL,
            due_date INTEGER,
            FOREIGN KEY(customer_id) REFERENCES customers(id) ON DELETE CASCADE
          )
        ''');
        await db.execute(
          'CREATE INDEX idx_txn_customer ON transactions(customer_id)',
        );
      },
    );
  }

  Future<List<Customer>> customers() async {
    final db = await _database;
    final rows = await db.query('customers', orderBy: 'name COLLATE NOCASE');
    return rows.map(Customer.fromMap).toList();
  }

  Future<Customer?> customer(int id) async {
    final db = await _database;
    final rows = await db.query('customers', where: 'id = ?', whereArgs: [id]);
    return rows.isEmpty ? null : Customer.fromMap(rows.first);
  }

  Future<int> insertCustomer(Customer c) async {
    final db = await _database;
    return db.insert('customers', c.toMap());
  }

  Future<void> updateCustomer(Customer c) async {
    final db = await _database;
    await db.update('customers', c.toMap(), where: 'id = ?', whereArgs: [c.id]);
  }

  Future<void> deleteCustomer(int id) async {
    final db = await _database;
    await db.delete('transactions', where: 'customer_id = ?', whereArgs: [id]);
    await db.delete('customers', where: 'id = ?', whereArgs: [id]);
  }

  Future<List<Txn>> allTxns() async {
    final db = await _database;
    final rows = await db.query('transactions');
    return rows.map(Txn.fromMap).toList();
  }

  Future<List<Txn>> txnsFor(int customerId) async {
    final db = await _database;
    final rows = await db.query(
      'transactions',
      where: 'customer_id = ?',
      whereArgs: [customerId],
      orderBy: 'date ASC, id ASC',
    );
    return rows.map(Txn.fromMap).toList();
  }

  Future<int> insertTxn(Txn t) async {
    final db = await _database;
    return db.insert('transactions', t.toMap());
  }

  Future<void> deleteTxn(int id) async {
    final db = await _database;
    await db.delete('transactions', where: 'id = ?', whereArgs: [id]);
  }

  // Membuat snapshot JSON tanpa mengubah database aktif.
  Future<Map<String, Object?>> exportBackup() async {
    final db = await _database;
    return {
      'format': backupFormat,
      'version': backupVersion,
      'exported_at': DateTime.now().toUtc().toIso8601String(),
      'customers': await db.query('customers', orderBy: 'id ASC'),
      'transactions': await db.query('transactions', orderBy: 'id ASC'),
    };
  }

  static List<Map<String, Object?>> _rows(
    Object? value,
    String label,
  ) {
    if (value is! List) {
      throw FormatException('Bagian $label bukan daftar yang valid');
    }

    return value.map((item) {
      if (item is! Map) {
        throw FormatException('Ada data $label yang tidak valid');
      }
      return Map<String, Object?>.from(item);
    }).toList();

import 'dart:convert';

import 'package:flutter_test/flutter_test.dart';
import 'package:buku_utang_warung/services/backup_crypto_service.dart';

void main() {
  final crypto = BackupCryptoService.instance;

  final sampleBackup = <String, Object?>{
    'format': 'buku_utang_warung_backup',
    'version': 1,
    'exported_at': '2026-11-14T00:00:00.000Z',
    'customers': [
      {'id': 1, 'name': 'Pelanggan Uji'},
    ],
    'transactions': [
      {'id': 1, 'customer_id': 1, 'amount': 25000},
    ],
  };

  test('Backup dapat dienkripsi dan dibuka kembali', () async {
    const password = 'WarungAman2026';

    final encrypted = await crypto.encryptBackup(
      sampleBackup,
      password,
    );

    expect(encrypted, isNotEmpty);
    expect(utf8.decode(encrypted), isNot(contains('Pelanggan Uji')));

    final restored = await crypto.decryptBackup(
      encrypted,
      password,
    );

    expect(restored['customers'], sampleBackup['customers']);
    expect(restored['transactions'], sampleBackup['transactions']);
  });

  test('Password kurang dari 8 karakter ditolak', () async {
    expect(
      () => crypto.encryptBackup(sampleBackup, '1234567'),
      throwsArgumentError,
    );
  });

  test('Password salah tidak dapat membuka backup', () async {
    final encrypted = await crypto.encryptBackup(
      sampleBackup,
      'PasswordBenar2026',
    );

    expect(
      () => crypto.decryptBackup(encrypted, 'PasswordSalah2026'),
      throwsFormatException,
    );
  });

  test('Isi ciphertext yang diubah akan ditolak', () async {
    final encrypted = await crypto.encryptBackup(
      sampleBackup,
      'PasswordBenar2026',
    );

    final envelope =
        jsonDecode(utf8.decode(encrypted)) as Map<String, dynamic>;

    final ciphertext = base64Decode(envelope['ciphertext'] as String);
    ciphertext[0] ^= 1;
    envelope['ciphertext'] = base64Encode(ciphertext);

    final tampered = utf8.encode(jsonEncode(envelope));

    expect(
      () => crypto.decryptBackup(tampered, 'PasswordBenar2026'),
      throwsFormatException,
    );
  });
}

import 'package:flutter_test/flutter_test.dart';

import 'package:buku_utang_warung/models.dart';
import 'package:buku_utang_warung/utils.dart';

Txn utang(int amount, DateTime date, {DateTime? due}) => Txn(
      customerId: 1,
      type: TxnType.utang,
      amount: amount,
      date: date,
      dueDate: due,
    );

Txn bayar(int amount, DateTime date) =>
    Txn(customerId: 1, type: TxnType.bayar, amount: amount, date: date);

void main() {
  test('rupiah memakai pemisah titik', () {
    expect(rupiah(0), 'Rp 0');
    expect(rupiah(500), 'Rp 500');
    expect(rupiah(15000), 'Rp 15.000');
    expect(rupiah(1250000), 'Rp 1.250.000');
    expect(rupiah(-2000), '-Rp 2.000');
  });

  test('tgl memakai nama bulan Indonesia', () {
    expect(tgl(DateTime(2026, 11, 5)), '5 Nov 2026');
    expect(tgl(DateTime(2026, 8, 17)), '17 Agu 2026');
  });

  test('normalizePhone ke format internasional', () {
    expect(normalizePhone('0812-3456-789'), '628123456789');
    expect(normalizePhone('+62 812 3456 789'), '628123456789');
    expect(normalizePhone('8123456789'), '628123456789');
    expect(normalizePhone(''), isNull);
  });

  test('saldo = utang dikurangi bayar', () {
    final txns = [
      utang(10000, DateTime(2026, 1, 1)),
      utang(20000, DateTime(2026, 1, 5)),
      bayar(5000, DateTime(2026, 1, 6)),
    ];
    expect(computeBalance(txns), 25000);
  });

  test('jatuh tempo dihitung dengan alokasi bayar ke utang terlama', () {
    final txns = [
      utang(10000, DateTime(2026, 1, 1), due: DateTime(2026, 1, 10)),
      utang(20000, DateTime(2026, 1, 5), due: DateTime(2026, 1, 20)),
      bayar(10000, DateTime(2026, 1, 6)),
    ];
    // Utang pertama sudah lunas, tinggal yang tempo 20 Jan.
    expect(nextDueDate(txns), DateTime(2026, 1, 20));
  });

  test('tanpa tempo atau sudah lunas -> tidak ada jatuh tempo', () {
    expect(nextDueDate([utang(1000, DateTime(2026, 1, 1))]), isNull);
    expect(
      nextDueDate([
        utang(1000, DateTime(2026, 1, 1), due: DateTime(2026, 1, 10)),
        bayar(1000, DateTime(2026, 1, 2)),
      ]),
      isNull,
    );
  });

  test('status pelanggan', () {
    final c = Customer(id: 1, name: 'Bu Ani', createdAt: DateTime(2026, 1, 1));
    final now = DateTime(2026, 11, 13, 9);

    CustomerSummary s(int balance, DateTime? due) =>
        CustomerSummary(customer: c, balance: balance, nextDue: due);

    expect(s(0, null).status(now), DebtStatus.lunas);
    expect(s(5000, null).status(now), DebtStatus.lancar);
    expect(s(5000, DateTime(2026, 11, 12)).status(now), DebtStatus.telat);
    expect(s(5000, DateTime(2026, 11, 13)).status(now), DebtStatus.mendekati);
    expect(s(5000, DateTime(2026, 11, 16)).status(now), DebtStatus.mendekati);
    expect(s(5000, DateTime(2026, 11, 20)).status(now), DebtStatus.lancar);
    expect(statusText(s(5000, DateTime(2026, 11, 10)), now), 'Telat 3 hari');
  });

  test('pesan tagih memuat nama dan nominal', () {
    final msg = buildReminder(
      name: 'Pak Budi',
      balance: 45000,
      due: DateTime(2026, 11, 20),
      txns: [utang(45000, DateTime(2026, 11, 1))],
    );
    expect(msg, contains('Pak Budi'));
    expect(msg, contains('Rp 45.000'));
    expect(msg, contains('20 Nov 2026'));
  });

  test('waUri menyandikan teks', () {
    final uri = waUri('08123456789', 'Halo dunia');
    expect(uri.host, 'wa.me');
    expect(uri.path, '/628123456789');
    expect(uri.queryParameters['text'], 'Halo dunia');
  });
}
